   257→<script define:vars={{ lang }}>
   258→  // DOM Elements
   259→  const form = document.getElementById('scheduler-form');
   260→  const phoneInput = document.getElementById('phone');
   261→  const messageInput = document.getElementById('message');
   262→  const dateInput = document.getElementById('date');
   263→  const timeInput = document.getElementById('time');
   264→  const charCount = document.getElementById('char-count');
   265→  const submitBtn = document.getElementById('submit-btn');
   266→  const successMessage = document.getElementById('success-message');
   267→  const successDetails = document.getElementById('success-details');
   268→  const taskLink = document.getElementById('task-link');
   269→  const copyLinkBtn = document.getElementById('copy-link-btn');
   270→  const scheduleAnotherBtn = document.getElementById('schedule-another');
   271→  const tasksList = document.getElementById('tasks-list');
   272→  const timezoneDisplay = document.getElementById('timezone-display');
   273→
   274→  // Set minimum date to today
   275→  const today = new Date().toISOString().split('T')[0];
   276→  dateInput.setAttribute('min', today);
   277→
   278→  // Display user's timezone
   279→  const userTimezone = Intl.DateTimeFormat().resolvedOptions().timeZone;
   280→  timezoneDisplay.textContent = `Timezone: ${userTimezone}`;
   281→
   282→  // Character counter
   283→  messageInput.addEventListener('input', () => {
   284→    charCount.textContent = messageInput.value.length;
   285→  });
   286→
   287→  // Format phone number as user types
   288→  phoneInput.addEventListener('input', (e) => {
   289→    let value = e.target.value.replace(/\D/g, '');
   290→    if (value.length > 15) value = value.slice(0, 15);
   291→    e.target.value = value;
   292→  });
   293→
   294→  // Form submission
   295→  form.addEventListener('submit', async (e) => {
   296→    e.preventDefault();
   297→    
   298→    const phone = phoneInput.value.replace(/\D/g, '');
   299→    const message = messageInput.value.trim();
   300→    const date = dateInput.value;
   301→    const time = timeInput.value;
   302→
   303→    // Validate
   304→    if (phone.length < 8 || phone.length > 15) {
   305→      alert('Please enter a valid phone number');
   306→      return;
   307→    }
   308→
   309→    const scheduledDateTime = new Date(`${date}T${time}`);
   310→    if (scheduledDateTime <= new Date()) {
   311→      alert('Please select a future date and time');
   312→      return;
   313→    }
   314→
   315→    // Show loading state
   316→    submitBtn.disabled = true;
   317→    submitBtn.innerHTML = '<span class="spinner"></span> Scheduling...';
   318→
   319→    try {
   320→      const response = await fetch('/api/scheduler', {
   321→        method: 'POST',
   322→        headers: { 'Content-Type': 'application/json' },
   323→        body: JSON.stringify({
   324→          phone,
   325→          message,
   326→          scheduledFor: scheduledDateTime.toISOString(),
   327→          timezone: userTimezone,
   328→          lang
   329→        })
   330→      });
   331→
   332→      const data = await response.json();
   333→
   334→      if (data.success) {
   335→        // Show success
   336→        form.classList.add('hidden');
   337→        successMessage.classList.remove('hidden');
   338→        
   339→        const formattedDate = new Date(data.task.scheduledFor).toLocaleString(lang);
   340→        successDetails.textContent = `${data.task.phone} at ${formattedDate}`;
   341→        
   342→        taskLink.href = `/scheduler/task/${data.task.id}`;
   343→        taskLink.textContent = `${window.location.origin}/scheduler/task/${data.task.id}`;
   344→        
   345→        // Save task ID to localStorage for listing
   346→        saveTaskLocally(data.task);
   347→        
   348→        // Refresh tasks list
   349→        loadTasks();
   350→      } else {
   351→        alert(data.error || 'Failed to schedule message');
   352→      }
   353→    } catch (error) {
   354→      console.error('Error:', error);
   355→      alert('Network error. Please try again.');
   356→    }
   357→
   358→    submitBtn.disabled = false;
   359→    submitBtn.innerHTML = `<svg class="btn-icon" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   360→      <rect x="3" y="4" width="18" height="18" rx="2"/>
   361→      <line x1="16" y1="2" x2="16" y2="6"/>
   362→      <line x1="8" y1="2" x2="8" y2="6"/>
   363→      <line x1="3" y1="10" x2="21" y2="10"/>
   364→    </svg> Schedule Message`;
   365→  });
   366→
   367→  // Copy link button
   368→  copyLinkBtn.addEventListener('click', async () => {
   369→    try {
   370→      await navigator.clipboard.writeText(taskLink.href);
   371→      copyLinkBtn.textContent = 'Copied!';
   372→      setTimeout(() => {
   373→        copyLinkBtn.textContent = copyLinkBtn.dataset.copy;
   374→      }, 2000);
   375→    } catch (err) {
   376→      console.error('Failed to copy:', err);
   377→    }
   378→  });
   379→
   380→  // Schedule another button
   381→  scheduleAnotherBtn.addEventListener('click', () => {
   382→    form.reset();
   383→    charCount.textContent = '0';
   384→    form.classList.remove('hidden');
   385→    successMessage.classList.add('hidden');
   386→  });
   387→
   388→  // Local storage functions for tasks
   389→  function saveTaskLocally(task) {
   390→    const tasks = JSON.parse(localStorage.getItem('opnchat-scheduled-tasks') || '[]');
   391→    tasks.unshift(task);
   392→    // Keep only last 50 tasks
   393→    if (tasks.length > 50) tasks.pop();
   394→    localStorage.setItem('opnchat-scheduled-tasks', JSON.stringify(tasks));
   395→  }
   396→
   397→  function loadTasks() {
   398→    const tasks = JSON.parse(localStorage.getItem('opnchat-scheduled-tasks') || '[]');
   399→    
   400→    if (tasks.length === 0) {
   401→      tasksList.innerHTML = `
   402→        <div class="empty-state">
   403→          <svg class="empty-icon" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5">
   404→            <rect x="3" y="4" width="18" height="18" rx="2"/>
   405→            <line x1="16" y1="2" x2="16" y2="6"/>
   406→            <line x1="8" y1="2" x2="8" y2="6"/>
   407→            <line x1="3" y1="10" x2="21" y2="10"/>
   408→            <line x1="9" y1="16" x2="15" y2="16"/>
   409→          </svg>
   410→          <p>No scheduled messages yet</p>
   411→        </div>
   412→      `;
   413→      return;
   414→    }
   415→
   416→    tasksList.innerHTML = tasks.map(task => {
   417→      const scheduledDate = new Date(task.scheduledFor);
   418→      const now = new Date();
   419→      const isPast = scheduledDate <= now;
   420→      const statusClass = isPast ? 'status-past' : 'status-pending';
   421→      const statusText = isPast ? 'Ready to send' : 'Scheduled';
   422→      
   423→      return `
   424→        <div class="task-item ${statusClass}">
   425→          <div class="task-info">
   426→            <span class="task-phone">${task.phone.replace(/(\d{1,3})(\d{3})(\d{3})(\d{2,})/, '+$1 $2 $3 $4')}</span>
   427→            <span class="task-time">${scheduledDate.toLocaleString(lang)}</span>
   428→          </div>
   429→          <div class="task-status">${statusText}</div>
   430→          <a href="/scheduler/task/${task.id}" class="task-view-btn" target="_blank">View</a>
   431→        </div>
   432→      `;
   433→    }).join('');
   434→  }
   435→
   436→  // Load tasks on page init
   437→  loadTasks();
   438→</script>
