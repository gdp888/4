   171→  // dies after the first SPA navigation because the element reference becomes stale.

   176→      var drawer = document.getElementById('nav-drawer');

   190→      if (!drawer || !overlay || !hamburger) return;

   197→    // Single delegated click listener — attached ONCE on document, which persists

   201→      if (e.target.closest('#nav-hamburger')) {
   202→        var drawer = document.getElementById('nav-drawer');

   204→        isOpen ? closeDrawer() : openDrawer();

   209→        closeDrawer();
   210→        return;
   211→      }
   212→      // Click on overlay (outside drawer) closes
   213→      if (e.target.closest('#nav-overlay')) {
   214→        closeDrawer();
   215→        return;
   216→      }
   217→      // Click on a nav link inside drawer closes (before VT navigation)
   218→      if (e.target.closest('#nav-drawer .nav-link')) {
   219→        closeDrawer();
   220→        return;
   221→      }
   222→    });
   223→
   224→    // Close on Escape — also delegated via document, survives VT swaps
   225→    document.addEventListener('keydown', function(e) {
   226→      if (e.key === 'Escape') closeDrawer();
   227→    });
   228→
   229→    // ===== Mobile Nav Search =====
   230→    (function() {
   231→      var searchInput = null;
   232→      var searchClear = null;
   233→      var categoriesWrapper = null;
   234→      var noResults = null;
   235→      var resultsInfo = null;
   236→
   237→      function initSearchElements() {
   238→        searchInput = document.getElementById('nav-search-input');
   239→        searchClear = document.getElementById('nav-search-clear');
   240→        categoriesWrapper = document.getElementById('nav-categories-wrapper');
   241→        noResults = document.getElementById('nav-no-results');
   242→        resultsInfo = document.getElementById('nav-search-results-info');
   243→      }
   244→
   245→      function filterNav(query) {
   246→        if (!searchInput || !categoriesWrapper) { initSearchElements(); }
   247→        if (!categoriesWrapper) return;
   248→
   249→        var q = query.toLowerCase().trim();
   250→        var categories = categoriesWrapper.querySelectorAll('[data-category]');
   251→        var totalVisible = 0;
   252→
   253→        // Show all categories wrapper, hide no-results
   254→        categoriesWrapper.style.display = '';
   255→        if (noResults) noResults.style.display = 'none';
   256→
   257→        if (!q) {
   258→          // Reset: show everything
   259→          categories.forEach(function(cat) {
   260→            cat.style.display = '';
   261→            cat.querySelectorAll('.nav-link').forEach(function(link) {
   262→              link.style.display = '';
   263→            });
   264→          });
   265→          if (searchClear) searchClear.style.display = 'none';
   266→          if (resultsInfo) resultsInfo.style.display = 'none';
   267→          return;
   268→        }
   269→
   270→        // Show clear button
   271→        if (searchClear) searchClear.style.display = 'flex';
   272→
   273→        categories.forEach(function(cat) {
   274→          var links = cat.querySelectorAll('[data-searchable]');
   275→          var visibleCount = 0;
   276→
   277→          links.forEach(function(link) {
   278→            var text = link.getAttribute('data-search-text') || '';
   279→            if (text.indexOf(q) !== -1) {
   280→              link.style.display = '';
   281→              visibleCount++;
   282→            } else {
   283→              link.style.display = 'none';
   284→            }
   285→          });
   286→
   287→          // Hide category header if no visible links
   288→          if (visibleCount > 0) {
   289→            cat.style.display = '';
   290→            totalVisible += visibleCount;
   291→          } else {
   292→            cat.style.display = 'none';
   293→          }
   294→        });
   295→
   296→        // Show/hide no results message
   297→        if (totalVisible === 0) {
   298→          categoriesWrapper.style.display = 'none';
   299→          if (noResults) noResults.style.display = 'flex';
   300→        }
   301→
   302→        // Show results count
   303→        if (resultsInfo && totalVisible > 0) {
   304→          resultsInfo.textContent = totalVisible + ' found';
   305→          resultsInfo.style.display = '';
   306→        } else if (resultsInfo) {
   307→          resultsInfo.style.display = 'none';
   308→        }
   309→      }
   310→
   311→      // Delegated events for search
   312→      document.addEventListener('input', function(e) {
   313→        if (e.target && e.target.id === 'nav-search-input') {
   314→          filterNav(e.target.value);
   315→        }
   316→      });
   317→
   318→      document.addEventListener('click', function(e) {
   319→        // Clear button
   320→        if (e.target.closest('#nav-search-clear')) {
   321→          if (!searchInput) initSearchElements();
   322→          if (searchInput) {
   323→            searchInput.value = '';
   324→            filterNav('');
   325→            searchInput.focus();
   326→          }
   327→          return;
   328→        }
   329→      });
   330→
   331→      // Auto-focus search when drawer opens
   332→      var originalOpen = openDrawer;
   333→      openDrawer = function() {
   334→        originalOpen();
   335→        setTimeout(function() {
   336→          if (!searchInput) initSearchElements();
   337→          if (searchInput) searchInput.focus();
   338→        }, 300); // After slide-in animation
   339→      };
   340→
   341→      // Clear search when drawer closes
   342→      var originalClose = closeDrawer;
   343→      closeClose = function() {
   344→        if (!searchInput) initSearchElements();
   345→        if (searchInput) {
   346→          searchInput.value = '';
   347→          filterNav('');
   348→        }
   349→        originalClose();
   350→      };
   351→      
   352→      // Fix: use the correct function name
   353→      closeDrawer = function() {
   354→        if (!searchInput) initSearchElements();
   355→        if (searchInput) {
   356→          searchInput.value = '';
   357→          filterNav('');
   358→        }
   359→        originalClose();
   360→      };
   361→    })();
   362→  })();
   363→</script>
