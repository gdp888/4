   300→  // Get hour in a timezone (0-23)
   301→  function getHourInTz(tz: string, date?: Date): number {
   302→    try {
   303→      const d = date || new Date();
   304→      const formatter = new Intl.DateTimeFormat(pageLocale, {
   305→        timeZone: tz, hour: 'numeric', hour12: false
   306→      });
   307→      const hour = parseInt(formatter.format(d));
   308→      return hour === 24 ? 0 : hour;
   309→    } catch { return 12; }
   310→  }

   312→  // Get offset in minutes for a timezone
   313→  function getOffsetMinutes(tz: string, date?: Date): number {
   314→    const d = date || new Date();
   315→    const utcDate = new Date(d.getTime());
   316→    const tzString = d.toLocaleString(pageLocale, { timeZone: tz });
   317→    const tzDate = new Date(tzString);
   318→    return (tzDate.getTime() - utcDate.getTime()) / 60000;
   319→  }

   321→  // Get offset label
   322→  function getOffsetLabel(tz: string): string {
   323→    try {
   324→      const formatter = new Intl.DateTimeFormat(pageLocale, {
   325→        timeZone: tz, timeZoneName: 'shortOffset'
   326→      });
   327→      const parts = formatter.formatToParts(new Date());
   328→      const offsetPart = parts.find(p => p.type === 'timeZoneName');
   329→      return offsetPart?.value || tz;
   330→    } catch { return tz; }
   331→  }

   343→  // Find overlap between two working hour ranges
   344→  function findOverlap(myTz: string, theirTz: string, workStart: number, workEnd: number): { myStart: number; myEnd: number; theirStart: number; theirEnd: number }[] {
   345→    const myOffset = getOffsetMinutes(myTz);
   346→    const theirOffset = getOffsetMinutes(theirTz);
   347→    const diffHours = (theirOffset - myOffset) / 60;
   348→
   349→    // Their working hours in my timezone
   350→    const theirWorkStartInMy = workStart - diffHours;
   351→    const theirWorkEndInMy = workEnd - diffHours;
   352→
   353→    const overlaps: { myStart: number; myEnd: number; theirStart: number; theirEnd: number }[] = [];
   354→
   355→    // Check today and tomorrow (wrap-around)
   356→    for (let dayOffset = -1; dayOffset <= 1; dayOffset++) {
   357→      const theirStartAdj = theirWorkStartInMy + dayOffset * 24;
   358→      const theirEndAdj = theirWorkEndInMy + dayOffset * 24;
   359→
   360→      const overlapStart = Math.max(workStart, theirStartAdj);
   361→      const overlapEnd = Math.min(workEnd, theirEndAdj);
   362→
   363→      if (overlapEnd > overlapStart && overlapStart >= 0 && overlapEnd <= 24) {
   364→        overlaps.push({
   365→          myStart: overlapStart,
   366→          myEnd: overlapEnd,
   367→          theirStart: overlapStart + diffHours,
   368→          theirEnd: overlapEnd + diffHours,
   369→        });
   370→      }
   371→    }
   372→
   373→    return overlaps;
   374→  }

   384→

   449→  function renderBestTime(overlaps: { myStart: number; myEnd: number; theirStart: number; theirEnd: number }[], myTz: string, theirTz: string, workStart: number, workEnd: number) {
   450→    if (!bestTimeContent) return;
   451→
   452→    const myName = cityNames[myTz] || myTz.split('/').pop() || myTz;
   453→    const theirName = cityNames[theirTz] || theirTz.split('/').pop() || theirTz;
   454→
   455→    if (overlaps.length === 0) {
   456→      bestTimeContent.innerHTML = `<div class="no-overlap">${i18n.noOverlap}</div>`;
   457→      return;
   458→    }
   459→
   460→    // Find the longest overlap
   461→    const longest = overlaps.reduce((a, b) => (b.myEnd - b.myStart) > (a.myEnd - a.myStart) ? b : a);
   462→    const duration = longest.myEnd - longest.myStart;
   463→    const midPoint = Math.floor((longest.myStart + longest.myEnd) / 2);
   464→
   465→    // Format times
   466→    const myStartStr = formatHour(longest.myStart);
   467→    const myEndStr = formatHour(longest.myEnd);
   468→    const theirStartStr = formatHour(longest.theirStart);
   469→    const theirEndStr = formatHour(longest.theirEnd);
   470→
   471→    bestTimeContent.innerHTML = `
   472→      <div class="best-time-main">
   473→        <div class="best-time-label">${myName}</div>
   474→        <div class="best-time-value">${myStartStr} — ${myEndStr}</div>
   475→        <div class="best-time-label">${theirName}</div>
   476→        <div class="best-time-value">${theirStartStr} — ${theirEndStr}</div>
   477→      </div>
   478→      <div class="best-time-duration">${duration}h ${i18n.overlapText}</div>
   479→    `;
   480→  }

   482→  function formatHour(h: number): string {
   483→    const normalized = ((h % 24) + 24) % 24;
   484→    const hours = Math.floor(normalized);
   485→    const minutes = Math.round((normalized - hours) * 60);
   486→    return `${String(hours).padStart(2, '0')}:${String(minutes).padStart(2, '0')}`;
   487→  }

   492→  function renderSlotPicker(overlaps: { myStart: number; myEnd: number; theirStart: number; theirEnd: number }[], myTz: string, theirTz: string) {
   493→    if (!slotGrid) return;
