   947→<script define:vars={{ lang }}>
   948→  const form = document.getElementById('telegram-form');

   950→  const messageInput = document.getElementById('message');
   951→  const errorMsg = document.getElementById('error-msg');
   952→  const historySection = document.getElementById('history-section');
   953→  const historyList = document.getElementById('history-list');
   954→  const clearBtn = document.getElementById('clear-history');
   955→
   956→  const STORAGE_KEY = 'opnchat-telegram-history';
   957→  const MAX_HISTORY = 5;
   958→
   959→  const localeMap = { en: 'en-US', es: 'es-ES', pt: 'pt-BR', id: 'id-ID' };
   960→  const locale = localeMap[lang] || 'en-US';
   961→
   962→  const chatLabel = 'Chat';
   963→
   964→  // Detect if input is a username or phone number
   965→  function isUsername(val) {
   966→    // Strip leading @, check if remaining is alphanumeric/underscores
   967→    var cleaned = val.replace(/^@/, '');
   968→    return /[a-zA-Z]/.test(cleaned);
   969→  }
   970→
   971→  // Clean username: strip @, keep alphanumeric + underscores
   972→  function cleanUsername(val) {
   973→    return val.replace(/^@/, '').replace(/[^a-zA-Z0-9_]/g, '');
   974→  }
   975→
   976→  // Clean phone: digits only
   977→  function cleanPhone(val) {
   978→    return val.replace(/\D/g, '');
   979→  }

   988→    try {
   989→      var data = localStorage.getItem(STORAGE_KEY);
   990→      return data ? JSON.parse(data) : [];
   991→    } catch {
   992→      return [];
   993→    }
   994→  }
   995→
   996→  function saveHistory(entry) {
   997→    var history = loadHistory().filter(h => h.value !== entry.value);
   998→    history.unshift(entry);
   999→    if (history.length > MAX_HISTORY) history.pop();
  1000→    localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
  1001→    renderHistory();
  1002→  }
  1003→
  1004→  function renderHistory() {

  1045→          : 'https://t.me/' + item.value;
  1046→        window.open(url, '_blank', 'noopener,noreferrer');
  1047→      });
  1048→
  1049→      li.appendChild(leftDiv);

  1074→      phone: lang === 'es' ? 'Ingresa un número de teléfono válido' : lang === 'pt' ? 'Digite um número de telefone válido' : lang === 'id' ? 'Silakan masukkan nomor telepon yang valid' : 'Please enter a valid phone number',
  1075→      short: lang === 'es' ? 'El número de teléfono es demasiado corto' : lang === 'pt' ? 'O número de telefone é muito curto' : lang === 'id' ? 'Nomor telepon terlalu pendek' : 'Phone number is too short',
  1076→    };
  1077→
  1078→    if (!rawVal) {
  1079→      errorMsg.textContent = errorMessages.username;
  1080→      input.focus();
  1081→      return;
  1082→    }
  1083→
  1084→    if (isUsername(rawVal)) {
  1085→      // Username mode
  1086→      var username = cleanUsername(rawVal);
  1087→      if (username.length < 3) {
  1088→        errorMsg.textContent = errorMessages.username;
  1089→        input.focus();
  1090→        return;

  1093→      var url = 'https://t.me/' + username;
