   433→  function getBaseOffset(tz: string, date: Date): number {
   434→    const utcDate = new Date(date.getTime());
   435→    const tzString = date.toLocaleString('en-US', { timeZone: tz });

   440→  function formatDiff(diffHours: number): string {
   441→    if (isNaN(diffHours)) return '';
   442→    if (diffHours === 0) return tzI18nEl?.dataset?.sameTime || 'Same time';
   443→    const sign = diffHours > 0 ? '+' : '-';
   444→    const abs = Math.abs(diffHours);
   445→    const h = Math.floor(abs);
   446→    const m = Math.round((abs - h) * 60);
   447→    if (m === 0) return sign + h + 'h';
   448→    if (m === 30) return sign + h + ':30h';
   449→    return sign + h + 'h ' + m + 'm';
   450→  }

   452→  function getTimeInTz(tz: string, baseTz: string, customTime?: string): { time: string; date: string; diff: string } {
   453→    try {
   454→      const now = new Date();
   455→      if (customTime) {
   456→        const [hours, minutes] = customTime.split(':').map(Number);
   457→        const baseFormatter = new Intl.DateTimeFormat(pageLocale, {
   458→          timeZone: baseTz, year: 'numeric', month: '2-digit', day: '2-digit',
   459→          hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: use12h ? true : false
   460→        });
   461→        const baseParts = baseFormatter.formatToParts(now);
   462→        const year = parseInt(baseParts.find(p => p.type === 'year')?.value || '0');
   463→        const month = parseInt(baseParts.find(p => p.type === 'month')?.value || '0') - 1;
   464→        const day = parseInt(baseParts.find(p => p.type === 'day')?.value || '0');
   465→        const utcDate = new Date(Date.UTC(year, month, day, hours, minutes, 0));
   466→        const baseOff = getBaseOffset(baseTz, utcDate);
   467→        utcDate.setUTCMinutes(utcDate.getUTCMinutes() - baseOff);
   468→        const targetFormatter = new Intl.DateTimeFormat(pageLocale, { timeZone: tz, hour: '2-digit', minute: '2-digit', hour12: use12h ? true : false });
   469→        const dateFormatter = new Intl.DateTimeFormat(pageLocale, { timeZone: tz, weekday: 'short', month: 'short', day: 'numeric' });
   470→        const time = targetFormatter.format(utcDate);
   471→        const dateStr = dateFormatter.format(utcDate);
   472→        const targetOffset = getBaseOffset(tz, utcDate);
   473→        const diffHours = (targetOffset - baseOff) / 60;
   474→        return { time, date: dateStr, diff: formatDiff(diffHours) };
   475→      } else {
   476→        const timeFormatter = new Intl.DateTimeFormat(pageLocale, { timeZone: tz, hour: '2-digit', minute: '2-digit', second: '2-digit', hour12: use12h ? true : false });
   477→        const dateFormatter = new Intl.DateTimeFormat(pageLocale, { timeZone: tz, weekday: 'short', month: 'short', day: 'numeric' });
   478→        const time = timeFormatter.format(now);
   479→        const dateStr = dateFormatter.format(now);
   480→        const baseOff = getBaseOffset(baseTz, now);
   481→        const targetOffset = getBaseOffset(tz, now);
   482→        const diffHours = (targetOffset - baseOff) / 60;
   483→        return { time, date: dateStr, diff: formatDiff(diffHours) };
   484→      }
   485→    } catch { return { time: '--:--', date: '', diff: '' }; }
   486→  }

   517→  function getBestHours(baseTz: string, allCities: string[]): number[] {
   518→    const bestHours: number[] = [];
   519→    for (let h = 0; h < 24; h++) {
   520→      const allWork = allCities.every(tz => {
   521→        const baseOffset = getBaseOffset(baseTz, new Date());
   522→        const targetOffset = getBaseOffset(tz, new Date());
   523→        const diffHours = (targetOffset - baseOffset) / 60;
   524→        let targetHour = (h + diffHours) % 24;
   525→        if (targetHour < 0) targetHour += 24;
   526→        return isWorkHour(Math.floor(targetHour));
   527→      });
   528→      if (allWork) bestHours.push(h);
   529→    }
   530→    return bestHours;
   531→  }

   533→  function mergeHoursToRanges(hours: number[]): { start: number; end: number }[] {
   534→    if (hours.length === 0) return [];
   535→    const ranges: { start: number; end: number }[] = [];
   536→    let rangeStart = hours[0], prev = hours[0];
   537→    for (let i = 1; i < hours.length; i++) {
   538→      if (hours[i] === prev + 1) { prev = hours[i]; }
   539→      else { ranges.push({ start: rangeStart, end: prev + 1 }); rangeStart = hours[i]; prev = hours[i]; }
   540→    }
   541→    ranges.push({ start: rangeStart, end: prev + 1 });
   542→    return ranges;
   543→  }

   551→  function filterSearchIndex(query: string): SearchEntry[] {
   552→    const q = query.trim().toLowerCase();
   553→    if (!q) return searchIndex.slice().sort((a, b) => a.localizedName.localeCompare(b.localizedName, undefined, { sensitivity: 'base' }));
   554→    const startsWith: SearchEntry[] = [], contains: SearchEntry[] = [];
   555→    for (const entry of searchIndex) {
   556→      if (entry.searchStr.includes(q)) {
   557→        if (entry.localizedName.toLowerCase().startsWith(q) || entry.englishName.toLowerCase().startsWith(q)) startsWith.push(entry);
   558→        else contains.push(entry);
   559→      }
   560→    }
   561→    const sortFn = (a: SearchEntry, b: SearchEntry) => a.localizedName.localeCompare(b.localizedName, undefined, { sensitivity: 'base' });
   562→    startsWith.sort(sortFn); contains.sort(sortFn);
   563→    return [...startsWith, ...contains];
   564→  }

   597→  function stepKeyboardNav(e: KeyboardEvent, container: HTMLElement | null, highlightedRef: { value: number }, onSelect: (tz: string) => void) {
   598→    if (!container) return;
   599→    const items = Array.from(container.querySelectorAll('.step-result')) as HTMLElement[];
   600→    if (e.key === 'ArrowDown') {
   601→      e.preventDefault();
   602→      if (highlightedRef.value < items.length - 1) highlightedRef.value++;
   603→      updateStepHighlight(container, highlightedRef.value);
   604→    } else if (e.key === 'ArrowUp') {
   605→      e.preventDefault();
   606→      if (highlightedRef.value > 0) highlightedRef.value--;
   607→      updateStepHighlight(container, highlightedRef.value);
   608→    } else if (e.key === 'Enter') {
   609→      e.preventDefault();
   610→      const item = items[highlightedRef.value];
   611→      if (item) onSelect(item.getAttribute('data-tz') || '');
   612→    } else if (e.key === 'Escape') {
   613→      e.preventDefault();
   614→      container.innerHTML = '';
   615→      highlightedRef.value = -1;
   616→    }
   617→  }
