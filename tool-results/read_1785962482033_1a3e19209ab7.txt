   868→   868→  const localeMap = { en: 'en-US', es: 'es-ES', pt: 'pt-BR', id: 'id-ID', ru: 'ru-RU', ar: 'ar-SA', de: 'de-DE' };
   869→   869→  const locale = localeMap[lang] || 'en-US';
   870→   870→
   871→   871→  const chatLabel = 'Viber';
   872→   872→
   873→   873→  // Phone input — digits only
   874→   874→  phoneInput.addEventListener('input', () => {
   875→   875→    let val = phoneInput.value.replace(/\D/g, '');
   876→   876→    if (val.length > 15) val = val.slice(0, 15);

   880→   880→
   881→   881→  // Load history from localStorage
   882→   882→  function loadHistory() {
   883→   883→    try {
   884→   884→      const data = localStorage.getItem(STORAGE_KEY);
   885→   885→      return data ? JSON.parse(data) : [];
   886→   886→    } catch {
   887→   887→      return [];
   888→   888→    }
   889→   889→  }
   890→   890→
   891→   891→  // Save — store clean digits (no +), display with +
   892→   892→  function saveHistory(cleanDigits) {
   893→   893→    const history = loadHistory().filter(h => h.number !== cleanDigits);
   894→   894→    history.unshift({ number: cleanDigits, timestamp: Date.now() });
   895→   895→    if (history.length > MAX_HISTORY) history.pop();
   896→   896→    localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
   897→   897→    renderHistory();
   898→   898→  }
   899→   899→
   900→   900→  function renderHistory() {
   901→   901→    const history = loadHistory();
   902→   902→    if (history.length === 0) {
   903→   903→      historySection.style.display = 'none';
   904→   904→      return;
   905→   905→    }
   906→   906→    historySection.style.display = 'block';
   907→   907→    historyList.innerHTML = '';
   908→   908→
   909→   909→    history.forEach(item => {
   910→   910→      const digits = item.number.replace(/\D/g, '');
   911→   911→      const li = document.createElement('li');
   912→   912→      li.className = 'history-item';
   913→   913→
   914→   914→      // Left side: number + time

   918→   918→      const numSpan = document.createElement('span');
   919→   919→      numSpan.className = 'history-number';
   920→   920→      numSpan.textContent = '+' + digits;
   921→   921→      // Click number → fill input
   922→   922→      numSpan.addEventListener('click', (e) => {
   923→   923→        e.stopPropagation();
   924→   924→        phoneInput.value = digits;
   925→   925→        phoneInput.focus();
   926→   926→      });
   927→   927→
   928→   928→      const timeSpan = document.createElement('span');
   929→   929→      timeSpan.className = 'history-time';
   930→   930→      const time = new Date(item.timestamp);
   931→   931→      timeSpan.textContent = time.toLocaleTimeString(locale, { hour: '2-digit', minute: '2-digit' });
   932→   932→
   933→   933→      leftDiv.appendChild(numSpan);
   934→   934→      leftDiv.appendChild(timeSpan);
   935→   935→
   936→   936→      // Right side: Chat button
   937→   937→      const chatBtn = document.createElement('button');

   942→   942→        e.stopPropagation();
   943→   943→        window.open(`viber://chat?number=%2B${digits}`, '_self');
   944→   944→      });
   945→   945→
   946→   946→      li.appendChild(leftDiv);
   947→   947→      li.appendChild(chatBtn);

   949→   949→    });
