   926→<script define:vars={{ lang }}>
   927→  const form = document.getElementById('whatsapp-form');
   928→  const phoneInput = document.getElementById('phone');
   929→  const messageInput = document.getElementById('message');
   930→  const errorMsg = document.getElementById('error-msg');
   931→  const historySection = document.getElementById('history-section');
   932→  const historyList = document.getElementById('history-list');
   933→  const clearBtn = document.getElementById('clear-history');
   934→
   935→  const STORAGE_KEY = 'opnchat-history';
   936→  const MAX_HISTORY = 5;
   937→
   938→  const localeMap = { en: 'en-US', es: 'es-ES', pt: 'pt-BR', id: 'id-ID' };
   939→  const locale = localeMap[lang] || 'en-US';
   940→
   941→  const chatLabel = lang === 'es' ? 'Chat' : lang === 'pt' ? 'Chat' : 'Chat';
   942→
   943→  // Phone input — digits only
   944→  phoneInput.addEventListener('input', () => {
   945→    let val = phoneInput.value.replace(/\D/g, '');
   946→    if (val.length > 15) val = val.slice(0, 15);
   947→    phoneInput.value = val;
   948→    errorMsg.textContent = '';
   949→  });
   950→
   951→  // Load history from localStorage
   952→  function loadHistory() {
   953→    try {
   954→      const data = localStorage.getItem(STORAGE_KEY);
   955→      return data ? JSON.parse(data) : [];
   956→    } catch {
   957→      return [];
   958→    }
   959→  }
   960→
   961→  // Save — store clean digits (no +), display with +
   962→  function saveHistory(cleanDigits) {
   963→    const history = loadHistory().filter(h => h.number !== cleanDigits);
   964→    history.unshift({ number: cleanDigits, timestamp: Date.now() });
   965→    if (history.length > MAX_HISTORY) history.pop();
   966→    localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
   967→    renderHistory();
   968→  }
   969→
   970→  function renderHistory() {
   971→    const history = loadHistory();
   972→    if (history.length === 0) {
   973→      historySection.style.display = 'none';
   974→      return;
   975→    }
   976→    historySection.style.display = 'block';
   977→    historyList.innerHTML = '';
   978→
   979→    history.forEach(item => {
   980→      const digits = item.number.replace(/\D/g, '');
   981→      const li = document.createElement('li');
   982→      li.className = 'history-item';
   983→
   984→      // Left side: number + time
   985→      const leftDiv = document.createElement('div');
   986→      leftDiv.className = 'history-left';
   987→
   988→      const numSpan = document.createElement('span');
   989→      numSpan.className = 'history-number';
   990→      numSpan.textContent = '+' + digits;
   991→      // Click number → fill input
   992→      numSpan.addEventListener('click', (e) => {
   993→        e.stopPropagation();
   994→        phoneInput.value = digits;
   995→        phoneInput.focus();
   996→      });
   997→
   998→      const timeSpan = document.createElement('span');
   999→      timeSpan.className = 'history-time';
  1000→      const time = new Date(item.timestamp);
  1001→      timeSpan.textContent = time.toLocaleTimeString(locale, { hour: '2-digit', minute: '2-digit' });
  1002→
  1003→      leftDiv.appendChild(numSpan);
  1004→      leftDiv.appendChild(timeSpan);
  1005→
  1006→      // Right side: Chat button
  1007→      const chatBtn = document.createElement('button');
  1008→      chatBtn.type = 'button';
  1009→      chatBtn.className = 'history-chat-btn';
  1010→      chatBtn.textContent = chatLabel;
  1011→      chatBtn.addEventListener('click', (e) => {
  1012→        e.stopPropagation();
  1013→        window.open(`https://wa.me/${digits}`, '_blank', 'noopener,noreferrer');
  1014→      });
  1015→
  1016→      li.appendChild(leftDiv);
  1017→      li.appendChild(chatBtn);
  1018→      historyList.appendChild(li);
  1019→    });
  1020→  }
  1021→
  1022→  // Clear history
  1023→  clearBtn.addEventListener('click', () => {
  1024→    localStorage.removeItem(STORAGE_KEY);
  1025→    renderHistory();
  1026→  });
  1027→
  1028→  // Initial render
  1029→  renderHistory();
  1030→
  1031→  // Form submit
  1032→  form.addEventListener('submit', (e) => {
  1033→    e.preventDefault();
  1034→    errorMsg.textContent = '';
  1035→
  1036→    const rawPhone = phoneInput.value.replace(/\D/g, '');
  1037→    const message = messageInput.value.trim();
  1038→
  1039→    const errorMessages = {
  1040→      valid: lang === 'es' ? 'Ingresa un número de teléfono válido' : lang === 'pt' ? 'Insira um número de telefone válido' : 'Please enter a valid phone number',
  1041→      short: lang === 'es' ? 'El número de teléfono es demasiado corto' : lang === 'pt' ? 'O número de telefone é muito curto' : 'Phone number is too short',
  1042→    };
  1043→
  1044→    if (rawPhone.length < 7) {
  1045→      errorMsg.textContent = errorMessages.valid;
  1046→      phoneInput.focus();
  1047→      return;
  1048→    }
  1049→
  1050→    if (rawPhone.length < 10) {
  1051→      errorMsg.textContent = errorMessages.short;
  1052→      phoneInput.focus();
  1053→      return;
  1054→    }
  1055→
  1056→    // Save clean digits (without +)
  1057→    saveHistory(rawPhone);
  1058→
  1059→    let url = `https://wa.me/${rawPhone}`;
  1060→    if (message) {
  1061→      url += `?text=${encodeURIComponent(message)}`;
  1062→    }
  1063→    window.open(url, '_blank');
  1064→  });
  1065→
  1066→  // Keyboard shortcut — Enter to submit when phone is focused
  1067→  phoneInput.addEventListener('keydown', (e) => {
  1068→    if (e.key === 'Enter') {
  1069→      e.preventDefault();
  1070→      form.dispatchEvent(new Event('submit', { cancelable: true }));
  1071→    }
  1072→  });
  1073→
  1074→  // Theme toggle
  1075→  const themeToggle = document.getElementById('theme-toggle');
  1076→  if (themeToggle) {
  1077→    themeToggle.addEventListener('click', () => {
  1078→      const current = document.documentElement.getAttribute('data-theme');
  1079→      const next = current === 'dark' ? 'light' : 'dark';
  1080→      document.documentElement.setAttribute('data-theme', next);
  1081→      localStorage.setItem('opnchat-theme', next);
  1082→    });
  1083→  }
