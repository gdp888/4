   327→        durationBtns: document.querySelectorAll('.duration-btn'),
   328→        workDuration: document.getElementById('work-duration'),
   329→        breakDuration: document.getElementById('break-duration'),
   330→        longBreakDuration: document.getElementById('long-break-duration'),
   331→        sessionsCount: document.getElementById('sessions-count'),
   332→        autoStartBreaksToggle: document.getElementById('auto-start-breaks'),
   333→        autoStartWorkToggle: document.getElementById('auto-start-work'),
   334→        soundEnabledToggle: document.getElementById('sound-enabled')
   335→      };
   336→      
   337→      this.loadState();
   338→      this.bindEvents();
   339→      this.updateDisplay();
   340→      this.updateSessionDots();
   341→      this.updateStats();
   342→    }
   343→    
   344→    bindEvents() {
   345→      // Main controls
   346→      this.elements.startBtn.addEventListener('click', () => this.toggle());
   347→      this.elements.resetBtn.addEventListener('click', () => this.reset());
   348→      this.elements.skipBtn.addEventListener('click', () => this.skip());
   349→      
   350→      // Mode tabs
   351→      this.elements.modeTabs.forEach(tab => {
   352→        tab.addEventListener('click', (e) => {
   353→          const mode = e.currentTarget.dataset.mode;
   354→          this.setMode(mode);
   355→        });
   356→      });
   357→      
   358→      // Task input
   359→      this.elements.addTaskBtn.addEventListener('click', () => this.addTask());
   360→      this.elements.taskInput.addEventListener('keypress', (e) => {
   361→        if (e.key === 'Enter') this.addTask();
   362→      });
   363→      this.elements.removeTaskBtn.addEventListener('click', () => this.removeTask());
   364→      
   365→      // Settings
   366→      this.elements.settingsToggle.addEventListener('click', () => this.toggleSettings());
   367→      this.elements.durationBtns.forEach(btn => {
   368→        btn.addEventListener('click', (e) => this.changeDuration(e.currentTarget));
   369→      });
   370→      
   371→      // Toggles
   372→      this.elements.autoStartBreaksToggle.addEventListener('change', (e) => {
   373→        this.autoStartBreaks = e.target.checked;
   374→        this.saveState();
   375→      });
   376→      this.elements.autoStartWorkToggle.addEventListener('change', (e) => {
   377→        this.autoStartWork = e.target.checked;
   378→        this.saveState();
   379→      });
   380→      this.elements.soundEnabledToggle.addEventListener('change', (e) => {
   381→        this.soundEnabled = e.target.checked;
   382→        this.saveState();
   383→      });
   384→      
   385→      // Keyboard shortcuts
   386→      document.addEventListener('keydown', (e) => {
   387→        // Don't trigger if typing in input
   388→        if (e.target.tagName === 'INPUT') return;
   389→        
   390→        switch(e.code) {
   391→          case 'Space':
   392→            e.preventDefault();
   393→            this.toggle();
   394→            break;
   395→          case 'KeyR':
   396→            this.reset();
   397→            break;
   398→          case 'KeyS':
   399→            this.skip();
   400→            break;
   401→        }
   402→      });
   403→      
   404→      // Visibility change - pause when tab hidden (optional)
   405→      document.addEventListener('visibilitychange', () => {
   406→        if (document.hidden && this.isRunning) {
   407→          // Keep running in background, just update title
   408→          this.updatePageTitle();
   409→        }
   410→      });
   411→    }
   412→    
   413→    toggle() {
   414→      if (this.isRunning) {
   415→        this.pause();
   416→      } else {
   417→        this.start();
   418→      }
   419→    }
   420→    
   421→    start() {
   422→      this.isRunning = true;
   423→      this.updatePlayButton();
   424→      
   425→      this.intervalId = setInterval(() => {
   426→        this.tick();
   427→      }, 1000);
   428→      
   429→      // Request wake lock on mobile
   430→      this.requestWakeLock();
   431→    }
   432→    
   433→    pause() {
   434→      this.isRunning = false;
   435→      this.updatePlayButton();
   436→      
   437→      if (this.intervalId) {
   438→        clearInterval(this.intervalId);
   439→        this.intervalId = null;
   440→      }
   441→      
   442→      this.releaseWakeLock();
   443→    }
   444→    
   445→    reset() {
   446→      this.pause();
   447→      this.timeRemaining = this.durations[this.mode];
   448→      this.updateDisplay();
   449→      this.saveState();
   450→    }
   451→    
   452→    skip() {
   453→      this.completeSession(true);
   454→    }
   455→    
   456→    tick() {
   457→      if (this.timeRemaining > 0) {
   458→        this.timeRemaining--;
   459→        this.updateDisplay();
   460→        this.updatePageTitle();
   461→        this.saveState();
   462→      } else {
   463→        this.completeSession(false);
   464→      }
   465→    }
   466→    
   467→    completeSession(skipped = false) {
   468→      this.pause();
   469→      this.playSound();
   470→      
   471→      if (this.mode === 'work' && !skipped) {
   472→        this.completedPomodoros++;
   473→        this.totalFocusTime += Math.round(this.durations.work / 60);
   474→        this.currentStreak++;
   475→        this.lastCompletedDate = new Date().toDateString();
   476→        
   477→        // Update session dot
   478→        const dot = this.elements.sessionDots?.querySelector(`[data-session="${this.currentSession}"]`);
   479→        if (dot) dot.classList.add('completed');
   480→        
   481→        this.currentSession++;
   482→        
   483→        if (this.currentSession > this.sessionsUntilLong) {
   484→          this.currentSession = 1;
   485→          this.setMode('longBreak');
   486→          // Reset all dots
   487→          this.updateSessionDots();
   488→        } else {
   489→          this.setMode('break');
   490→        }
   491→        
   492→        if (this.autoStartBreaks && !skipped) {
   493→          setTimeout(() => this.start(), 1000);
   494→        }
   495→      } else {
   496→        this.setMode('work');
   497→        if (this.autoStartWork && !skipped) {
   498→          setTimeout(() => this.start(), 1000);
   499→        }
   500→      }
   501→      
   502→      this.updateStats();
   503→      this.showNotification();
   504→      this.saveState();
   505→    }
   506→    
   507→    setMode(mode) {
   508→      this.mode = mode;
   509→      this.timeRemaining = this.durations[mode];
   510→      this.isRunning = false;
   511→      
   512→      // Update tabs
   513→      this.elements.modeTabs.forEach(tab => {
   514→        tab.classList.toggle('active', tab.dataset.mode === mode);
   515→      });
   516→      
   517→      // Update mode label with i18n
   518→      const modeLabels = {
   519→        work: this.getI18n('pomodoro.mode.work') || 'Focus',
   520→        break: this.getI18n('pomodoro.mode.break') || 'Break',
   521→        longBreak: this.getI18n('pomodoro.mode.longBreak') || 'Long Break'
   522→      };
   523→      if (this.elements.modeLabel) {
   524→        this.elements.modeLabel.textContent = modeLabels[mode];
   525→      }
   526→      
   527→      // Update progress color
   528→      if (this.elements.timerProgress) {
   529→        this.elements.timerProgress.style.stroke = this.getModeColor(mode);
   530→      }
   531→      
   532→      this.updateDisplay();
   533→      this.updatePlayButton();
   534→      this.saveState();
   535→    }
   536→    
   537→    getModeColor(mode) {
   538→      const colors = {
   539→        work: '#E53935',
   540→        break: '#25D366',
   541→        longBreak: '#5C6BC0'
   542→      };
   543→      return colors[mode] || colors.work;
   544→    }
   545→    
   546→    updateDisplay() {
   547→      const minutes = Math.floor(this.timeRemaining / 60);
   548→      const seconds = this.timeRemaining % 60;
   549→      const timeString = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
   550→      
   551→      if (this.elements.timeDisplay) {
   552→        this.elements.timeDisplay.textContent = timeString;
   553→      }
   554→      
   555→      // Update circular progress
   556→      if (this.elements.timerProgress) {
   557→        const totalTime = this.durations[this.mode];
   558→        const progress = this.timeRemaining / totalTime;
   559→        const circumference = 2 * Math.PI * 120; // r=120
   560→        const offset = circumference * (1 - progress);
   561→        this.elements.timerProgress.style.strokeDashoffset = offset;
   562→      }
   563→      
   564→      // Update page title
   565→      this.updatePageTitle();
   566→    }
   567→    
   568→    updatePageTitle() {
   569→      const minutes = Math.floor(this.timeRemaining / 60);
   570→      const seconds = this.timeRemaining % 60;
   571→      const prefix = this.isRunning ? '⏱' : '🍅';
   572→      document.title = `${prefix} ${minutes}:${seconds.toString().padStart(2, '0')} — OPN Chat`;
   573→    }
   574→    
   575→    updatePlayButton() {
   576→      const playIcon = this.elements.startBtn?.querySelector('.play-icon');
   577→      const pauseIcon = this.elements.startBtn?.querySelector('.pause-icon');
   578→      
   579→      if (playIcon && pauseIcon) {
   580→        playIcon.style.display = this.isRunning ? 'none' : 'block';
   581→        pauseIcon.style.display = this.isRunning ? 'block' : 'none';
   582→      }
   583→    }
   584→    
   585→    updateSessionDots() {
   586→      if (!this.elements.sessionDots) return;
   587→      
   588→      const dots = this.elements.sessionDots.querySelectorAll('.dot');
   589→      dots.forEach((dot, index) => {
   590→        dot.classList.toggle('completed', index < (this.currentSession - 1));
   591→      });
   592→    }
   593→    
   594→    updateStats() {
   595→      if (this.elements.completedCount) {
   596→        this.elements.completedCount.textContent = this.completedPomodoros;
   597→      }
   598→      if (this.elements.totalTime) {
   599→        this.elements.totalTime.textContent = `${this.totalFocusTime}m`;
   600→      }
   601→      if (this.elements.currentStreak) {
   602→        // Check if streak should reset (new day)
   603→        const today = new Date().toDateString();
   604→        if (this.lastCompletedDate && this.lastCompletedDate !== today && this.completedPomodoros > 0) {
   605→          // Streak continues until next day without completion
   606→        }
   607→        this.elements.currentStreak.textContent = `🔥 ${this.currentStreak}`;
   608→      }
   609→    }
   610→    
   611→    addTask() {
   612→      const text = this.elements.taskInput.value.trim();
   613→      if (!text) return;
   614→      
   615→      if (this.elements.activeTaskText) {
   616→        this.elements.activeTaskText.textContent = text;
   617→      }
   618→      if (this.elements.activeTask) {
   619→        this.elements.activeTask.style.display = 'flex';
   620→      }
   621→      this.elements.taskInput.value = '';
   622→      this.saveState();
   623→    }
   624→    
   625→    removeTask() {
   626→      if (this.elements.activeTaskText) {
   627→        this.elements.activeTaskText.textContent = '';
   628→      }
   629→      if (this.elements.activeTask) {
   630→        this.elements.activeTask.style.display = 'none';
   631→      }
   632→      this.saveState();
   633→    }
   634→    
   635→    toggleSettings() {
   636→      if (!this.elements.settingsPanel) return;
   637→      
   638→      const isVisible = this.elements.settingsPanel.style.display !== 'none';
   639→      this.elements.settingsPanel.style.display = isVisible ? 'none' : 'block';
   640→    }
   641→    
   642→    changeDuration(btn) {
   643→      const setting = btn.dataset.setting;
   644→      const change = parseInt(btn.dataset.change);
   645→      
   646→      switch(setting) {
   647→        case 'work':
   648→          this.durations.work = Math.max(60, Math.max(1, this.durations.work / 60 + change)) * 60;
   649→          if (this.elements.workDuration) this.elements.workDuration.textContent = this.durations.work / 60;
   650→          break;
   651→        case 'break':
   652→          this.durations.break = Math.max(60, Math.max(1, this.durations.break / 60 + change)) * 60;
   653→          if (this.elements.breakDuration) this.elements.breakDuration.textContent = this.durations.break / 60;
   654→          break;
   655→        case 'longBreak':
   656→          this.durations.longBreak = Math.max(60, Math.max(1, this.durations.longBreak / 60 + change)) * 60;
   657→          if (this.elements.longBreakDuration) this.elements.longBreakDuration.textContent = this.durations.longBreak / 60;
   658→          break;
   659→        case 'sessions':
   660→          this.sessionsUntilLong = Math.max(2, Math.min(10, this.sessionsUntilLong + change));
   661→          if (this.elements.sessionsCount) this.elements.sessionsCount.textContent = this.sessionsUntilLong;
   662→          this.updateSessionDots();
   663→          break;
   664→      }
   665→      
   666→      if (this.mode === setting || (setting === 'sessions')) {
   667→        this.timeRemaining = this.durations[this.mode];
   668→        this.updateDisplay();
   669→      }
   670→      
   671→      this.saveState();
   672→    }
   673→    
   674→    // ===== AUDIO =====
   675→    playSound() {
   676→      if (!this.soundEnabled) return;
   677→      
   678→      try {
   679→        if (!this.audioContext) {
   680→          this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
   681→        }
   682→        
   683→        const oscillator = this.audioContext.createOscillator();
   684→        const gainNode = this.audioContext.createGain();
   685→        
   686→        oscillator.connect(gainNode);
   687→        gainNode.connect(this.audioContext.destination);
   688→        
   689→        oscillator.frequency.setValueAtTime(800, this.audioContext.currentTime);
   690→        oscillator.type = 'sine';
   691→        
   692→        gainNode.gain.setValueAtTime(0.3, this.audioContext.currentTime);
   693→        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.5);
   694→        
   695→        oscillator.start(this.audioContext.currentTime);
   696→        oscillator.stop(this.audioContext.currentTime + 0.5);
   697→      } catch (e) {
   698→        console.log('Audio not available:', e);
   699→      }
   700→    }
   701→    
   702→    showNotification() {
   703→      if ('Notification' in window && Notification.permission === 'granted') {
   704→        const messages = {
   705→          work: this.getI18n('pomodoro.notification.workDone') || 'Time to take a break!',
   706→          break: this.getI18n('pomodoro.notification.breakDone') || 'Time to focus!',
   707→          longBreak: this.getI18n('pomodoro.notification.breakDone') || 'Time to focus!'
   708→        };
   709→        
   710→        new Notification('🍅 OPN Chat', {
   711→          body: messages[this.mode],
   712→          icon: '/favicon.ico'
   713→        });
   714→      }
   715→    }
   716→    
   717→    requestNotificationPermission() {
   718→      if ('Notification' in window && Notification.permission === 'default') {
   719→        Notification.requestPermission();
   720→      }
   721→    }
   722→    
   723→    // ===== WAKE LOCK (Mobile) =====
   724→    async function getWakeLock() {
   725→      try {
   726→        if ('wakeLock' in navigator) {
   727→          return await navigator.wakeLock.request('screen');
   728→        }
   729→      } catch (err) {
   730→        console.log('Wake Lock error:', err);
   731→      }
   732→      return null;
   733→    }
   734→    
   735→    let wakeLock = null;
   736→    
   737→    async requestWakeLock() {
   738→      wakeLock = await getWakeLock();
   739→    }
   740→    
   741→    releaseWakeLock() {
   742→      if (wakeLock) {
   743→        wakeLock.release();
   744→        wakeLock = null;
   745→      }
   746→    }
   747→    
   748→    // Make methods available on instance
   749→    this.requestWakeLock = requestWakeLock;
   750→    this.releaseWakeLock = releaseWakeLock;
   751→    
   752→    // ===== STATE PERSISTENCE =====
   753→    saveState() {
   754→      const state = {
   755→        mode: this.mode,
   756→        timeRemaining: this.timeRemaining,
   757→        isRunning: this.isRunning,
   758→        currentSession: this.currentSession,
   759→        completedPomodoros: this.completedPomodoros,
   760→        totalFocusTime: this.totalFocusTime,
   761→        currentStreak: this.currentStreak,
   762→        lastCompletedDate: this.lastCompletedDate,
   763→        durations: this.durations,
   764→        sessionsUntilLong: this.sessionsUntilLong,
   765→        autoStartBreaks: this.autoStartBreaks,
   766→        autoStartWork: this.autoStartWork,
   767→        soundEnabled: this.soundEnabled,
   768→        taskText: this.elements.activeTaskText?.textContent || ''
   769→      };
   770→      
   771→      try {
   772→        localStorage.setItem('pomodoro-state', JSON.stringify(state));
   773→      } catch (e) {
   774→        console.log('Could not save state:', e);
   775→      }
   776→    }
   777→    
   778→    loadState() {
   779→      try {
   780→        const saved = localStorage.getItem('pomodoro-state');
   781→        if (saved) {
   782→          const state = JSON.parse(saved);
   783→          
   784→          // Only restore if saved today
   785→          if (state.lastCompletedDate) {
   786→            const savedDate = new Date(state.lastCompletedDate);
   787→            const today = new Date();
   788→            if (savedDate.toDateString() !== today.toDateString()) {
   789→              // New day - reset daily stats but keep settings
   790→              this.durations = state.durations || this.durations;
   791→              this.sessionsUntilLong = state.sessionsUntilLong || this.sessionsUntilLong;
   792→              this.autoStartBreaks = state.autoStartBreaks ?? this.autoStartBreaks;
   793→              this.autoStartWork = state.autoStartWork ?? this.autoStartWork;
   794→              this.soundEnabled = state.soundEnabled ?? this.soundEnabled;
   795→              return;
   796→            }
   797→          }
   798→          
   799→          Object.assign(this, state);
   800→          
   801→          // Restore UI after a short delay for DOM
   802→          setTimeout(() => {
   803→            this.updateDisplay();
   804→            this.updateSessionDots();
   805→            this.updateStats();
   806→            this.updatePlayButton();
   807→            
   808→            // Restore task
   809→            if (state.taskText && this.elements.activeTaskText) {
   810→              this.elements.activeTaskText.textContent = state.taskText;
   811→              if (this.elements.activeTask) {
   812→                this.elements.activeTask.style.display = 'flex';
   813→              }
   814→            }
   815→            
   816→            // Restore settings UI
   817→            if (this.elements.workDuration) this.elements.workDuration.textContent = this.durations.work / 60;
   818→            if (this.elements.breakDuration) this.elements.breakDuration.textContent = this.durations.break / 60;
   819→            if (this.elements.longBreakDuration) this.elements.longBreakDuration.textContent = this.durations.longBreak / 60;
   820→            if (this.elements.sessionsCount) this.elements.sessionsCount.textContent = this.sessionsUntilLong;
   821→            if (this.elements.autoStartBreaksToggle) this.elements.autoStartBreaksToggle.checked = this.autoStartBreaks;
   822→            if (this.elements.autoStartWorkToggle) this.elements.autoStartWorkToggle.checked = this.autoStartWork;
   823→            if (this.elements.soundEnabledToggle) this.elements.soundEnabledToggle.checked = this.soundEnabled;
   824→            
   825→            // Update mode tabs
   826→            this.elements.modeTabs.forEach(tab => {
   827→              tab.classList.toggle('active', tab.dataset.mode === this.mode);
   828→            });
   829→            
   830→            // Update progress color
   831→            if (this.elements.timerProgress) {
   832→              this.elements.timerProgress.style.stroke = this.getModeColor(this.mode);
   833→            }
   834→            
   835→            // Resume if was running
   836→            if (this.isRunning) {
   837→              this.start();
   838→            }
   839→          }, 100);
   840→        }
   841→      } catch (e) {
   842→        console.log('Could not load state:', e);
   843→      }
   844→      
   845→      // Request notification permission on first interaction
   846→      document.addEventListener('click', () => this.requestNotificationPermission(), { once: true });
   847→    }
   848→    
   849→    // Helper to get i18n strings from data attributes
   850→    getI18n(key) {
   851→      const el = document.querySelector(`[data-i18n="${key}"]`);
   852→      return el?.textContent || null;
   853→    }
   854→  }

   857→  const timer = new PomodoroTimer();
   858→})();
