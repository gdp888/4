   386→    // Generate link
   387→    function doGenerate() {
   388→      var username = usernameInput.value.trim().replace(/^@/, '');
   389→      var message = messageInput.value.trim();
   390→
   391→      if (!username) {
   392→        errorMsg.textContent = usernameInput.getAttribute('data-error-empty');
   393→        errorMsg.style.display = 'block';
   394→        return;
   395→      }
   396→
   397→      // Validate username: 5-32 chars, letters, digits, underscores
   398→      if (!/^[a-zA-Z][a-zA-Z0-9_]{4,31}$/.test(username)) {
   399→        errorMsg.textContent = usernameInput.getAttribute('data-error-format');
   400→        errorMsg.style.display = 'block';
   401→        return;
   402→      }
   403→
   404→      errorMsg.style.display = 'none';
   405→
   406→      var link = 'https://t.me/' + username;
   407→      if (message) {
   408→        if (selectedLinkType === 'bot') {
   409→          link += '?start=' + encodeURIComponent(message);
   410→        } else {
   411→          link += '?text=' + encodeURIComponent(message);
   412→        }
   413→      }
   414→
   415→      generatedLink = link;
   416→      linkDisplay.textContent = link;
   417→      resultSection.style.display = 'block';
   418→      previewMsg.textContent = message || '👋';
   419→      previewMsg.style.display = 'block';
   420→
   421→      var qrCanvas = document.getElementById('tgl-qr-canvas');
   422→      generateQR(link, qrCanvas);
   423→
   424→      resultSection.scrollIntoView({ behavior: 'smooth', block: 'start' });
   425→    }
   426→
   427→    // Copy link
   428→    function doCopy(text) {
   429→      navigator.clipboard.writeText(text).then(function() {
   430→        var label = copyBtn.querySelector('.copy-label');
   431→        var originalText = label.textContent;
   432→        label.textContent = '✓';
   433→        setTimeout(function() { label.textContent = originalText; }, 1500);
   434→      }).catch(function() {
   435→        var textarea = document.createElement('textarea');
   436→        textarea.value = text;
   437→        document.body.appendChild(textarea);
   438→        textarea.select();
   439→        document.execCommand('copy');
   440→        document.body.removeChild(textarea);
   441→      });
   442→    }
   443→
   444→    // Download QR
   445→    function doDownloadQR() {
   446→      var qrCanvas = document.getElementById('tgl-qr-canvas');
   447→      var canvas = qrCanvas && qrCanvas.querySelector('canvas');
   448→      if (canvas) {
   449→        var link = document.createElement('a');
   450→        link.download = 'telegram-link-qr.png';
   451→        link.href = canvas.toDataURL('image/png');
   452→        link.click();
   453→      } else {
   454→        var img = qrCanvas && qrCanvas.querySelector('img');
   455→        if (img) {
   456→          var link = document.createElement('a');
   457→          link.download = 'telegram-link-qr.png';
   458→          link.href = img.src;
   459→          link.click();
   460→        }
   461→      }
   462→    }
