   166→<script is:inline>
   167→  // Mobile nav drawer toggle
   168→  // IMPORTANT: Use event delegation on `document` so the listener survives
   169→  // Astro View Transitions (ClientRouter) DOM swaps. Direct element.addEventListener
   170→  // dies after the first SPA navigation because the element reference becomes stale.
   171→  // The lang switcher and theme toggle already use this pattern (see LanguageSwitcher.astro
   172→  // and Layout.astro) — that's why they work after VT navigation but the burger didn't.
   173→  (function() {
   174→    function openDrawer() {
   175→      var drawer = document.getElementById('nav-drawer');
   176→      var overlay = document.getElementById('nav-overlay');
   177→      var hamburger = document.getElementById('nav-hamburger');
   178→      if (!drawer || !overlay || !hamburger) return;
   179→      drawer.classList.add('nav-drawer-open');
   180→      overlay.classList.add('nav-overlay-open');
   181→      hamburger.setAttribute('aria-expanded', 'true');
   182→      document.body.style.overflow = 'hidden';
   183→    }
   184→
   185→    function closeDrawer() {
   186→      var drawer = document.getElementById('nav-drawer');
   187→      var overlay = document.getElementById('nav-overlay');
   188→      var hamburger = document.getElementById('nav-hamburger');
   189→      if (!drawer || !overlay || !hamburger) return;
   190→      drawer.classList.remove('nav-drawer-open');
   191→      overlay.classList.remove('nav-overlay-open');
   192→      hamburger.setAttribute('aria-expanded', 'false');
   193→      document.body.style.overflow = '';
   194→    }
   195→
   196→    // Single delegated click listener — attached ONCE on document, which persists
   197→    // across View Transitions swaps. Elements are looked up fresh inside handlers.
   198→    document.addEventListener('click', function(e) {
   199→      // Burger toggle
   200→      if (e.target.closest('#nav-hamburger')) {
   201→        var drawer = document.getElementById('nav-drawer');
   202→        var isOpen = drawer && drawer.classList.contains('nav-drawer-open');
   203→        isOpen ? closeDrawer() : openDrawer();
   204→        return;
   205→      }
   206→      // Close button inside drawer
   207→      if (e.target.closest('#nav-drawer-close')) {
   208→        closeDrawer();
   209→        return;
   210→      }
   211→      // Click on overlay (outside drawer) closes
   212→      if (e.target.closest('#nav-overlay')) {
   213→        closeDrawer();
   214→        return;
   215→      }
   216→      // Click on a nav link inside drawer closes (before VT navigation)
   217→      if (e.target.closest('#nav-drawer .nav-link')) {
   218→        closeDrawer();
   219→        return;
   220→      }
   221→    });
   222→
   223→    // Close on Escape — also delegated via document, survives VT swaps
   224→    document.addEventListener('keydown', function(e) {
   225→      if (e.key === 'Escape') closeDrawer();
   226→    });
   227→
   228→    // ===== Mobile Nav Search =====
   229→    (function() {
   230→      var searchInput = null;
   231→      var searchClear = null;
   232→      var categoriesWrapper = null;
   233→      var noResults = null;
   234→      var resultsInfo = null;
   235→
   236→      function initSearchElements() {
   237→        searchInput = document.getElementById('nav-search-input');
   238→        searchClear = document.getElementById('nav-search-clear');
   239→        categoriesWrapper = document.getElementById('nav-categories-wrapper');
   240→        noResults = document.getElementById('nav-no-results');
   241→        resultsInfo = document.getElementById('nav-search-results-info');
   242→      }
   243→
   244→      function filterNav(query) {
   245→        if (!searchInput || !categoriesWrapper) { initSearchElements(); }
   246→        if (!categoriesWrapper) return;
   247→
   248→        var q = query.toLowerCase().trim();
   249→        var categories = categoriesWrapper.querySelectorAll('[data-category]');
   250→        var totalVisible = 0;
   251→
   252→        // Show all categories wrapper, hide no-results
   253→        categoriesWrapper.style.display = '';
   254→        if (noResults) noResults.style.display = 'none';
   255→
   256→        if (!q) {
   257→          // Reset: show everything
   258→          categories.forEach(function(cat) {
   259→            cat.style.display = '';
   260→            cat.querySelectorAll('.nav-link').forEach(function(link) {
   261→              link.style.display = '';
   262→            });
   263→          });
   264→          if (searchClear) searchClear.style.display = 'none';
   265→          if (resultsInfo) resultsInfo.style.display = 'none';
   266→          return;
   267→        }
   268→
   269→        // Show clear button
   270→        if (searchClear) searchClear.style.display = 'flex';
   271→
   272→        categories.forEach(function(cat) {
   273→          var links = cat.querySelectorAll('[data-searchable]');
   274→          var visibleCount = 0;
   275→
   276→          links.forEach(function(link) {
   277→            var text = link.getAttribute('data-search-text') || '';
   278→            if (text.indexOf(q) !== -1) {
   279→              link.style.display = '';
   280→              visibleCount++;
   281→            } else {
   282→              link.style.display = 'none';
   283→            }
   284→          });
   285→
   286→          // Hide category header if no visible links
   287→          if (visibleCount > 0) {
   288→            cat.style.display = '';
   289→            totalVisible += visibleCount;
   290→          } else {
   291→            cat.style.display = 'none';
   292→          }
   293→        });
   294→
   295→        // Show/hide no results message
   296→        if (totalVisible === 0) {
   297→          categoriesWrapper.style.display = 'none';
   298→          if (noResults) noResults.style.display = 'flex';
   299→        }
   300→
   301→        // Show results count
   302→        if (resultsInfo && totalVisible > 0) {
   303→          resultsInfo.textContent = totalVisible + ' found';
   304→          resultsInfo.style.display = '';
   305→        } else if (resultsInfo) {
   306→          resultsInfo.style.display = 'none';
   307→        }
   308→      }
   309→
   310→      // Delegated events for search
   311→      document.addEventListener('input', function(e) {
   312→        if (e.target && e.target.id === 'nav-search-input') {
   313→          filterNav(e.target.value);
   314→        }
   315→      });
   316→
   317→      document.addEventListener('click', function(e) {
   318→        // Clear button
   319→        if (e.target.closest('#nav-search-clear')) {
   320→          if (!searchInput) initSearchElements();
   321→          if (searchInput) {
   322→            searchInput.value = '';
   323→            filterNav('');
   324→            searchInput.focus();
   325→          }
   326→          return;
   327→        }
   328→      });
   329→
   330→      // Auto-focus search when drawer opens
   331→      var originalOpen = openDrawer;
   332→      openDrawer = function() {
   333→        originalOpen();
   334→        setTimeout(function() {
   335→          if (!searchInput) initSearchElements();
   336→          if (searchInput) searchInput.focus();
   337→        }, 300); // After slide-in animation
   338→      };
   339→
   340→      // Clear search when drawer closes
   341→      var originalClose = closeDrawer;
   342→      closeClose = function() {
   343→        if (!searchInput) initSearchElements();
   344→        if (searchInput) {
   345→          searchInput.value = '';
   346→          filterNav('');
   347→        }
   348→        originalClose();
   349→      };
   350→      
   351→      // Fix: use the correct function name
   352→      closeDrawer = function() {
   353→        if (!searchInput) initSearchElements();
   354→        if (searchInput) {
   355→          searchInput.value = '';
   356→          filterNav('');
   357→        }
   358→        originalClose();
   359→      };
   360→    })();
   361→  })();
   362→</script>
