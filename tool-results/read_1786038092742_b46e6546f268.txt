   294→<script is:inline>
   295→(function() {
   296→  // ===== POMODORO TIMER CLASS =====
   297→  class PomodoroTimer {
   298→    constructor() {
   299→      // Default settings (in minutes/seconds)
   300→      this.durations = {
   301→        work: 25 * 60,
   302→        break: 5 * 60,
   303→        longBreak: 15 * 60
   304→      };
   305→      this.sessionsUntilLong = 4;
   306→      
   307→      // State
   308→      this.mode = 'work'; // work | break | longBreak
   309→      this.timeRemaining = this.durations.work;
   310→      this.isRunning = false;
   311→      this.intervalId = null;
   312→      this.currentSession = 1;
   313→      this.completedPomodoros = 0;
   314→      this.totalFocusTime = 0; // in minutes
   315→      this.currentStreak = 0;
   316→      this.lastCompletedDate = null;
   317→      
   318→      // Settings
   319→      this.autoStartBreaks = true;
   320→      this.autoStartWork = false;
   321→      this.soundEnabled = true;
   322→      
   323→      // DOM Elements (initialized after DOM ready)
   324→      this.elements = {};
   325→      
   326→      // Audio context for notification sound
   327→      this.audioContext = null;
   328→      
   329→      this.init();
   330→    }
   331→    
   332→    init() {
   333→      // Wait for DOM to be ready
   334→      if (document.readyState === 'loading') {
   335→        document.addEventListener('DOMContentLoaded', () => this.setupDOM());
   336→      } else {
   337→        this.setupDOM();
   338→      }
   339→    }
   340→    
   341→    setupDOM() {
   342→      // Cache DOM elements
   343→      this.elements = {
   344→        timeDisplay: document.getElementById('time-display'),
   345→        modeLabel: document.getElementById('mode-label'),
   346→        startBtn: document.getElementById('start-btn'),
   347→        resetBtn: document.getElementById('reset-btn'),
   348→        skipBtn: document.getElementById('skip-btn'),
   349→        sessionDots: document.getElementById('session-dots'),
   350→        completedCount: document.getElementById('completed-count'),
   351→        totalTime: document.getElementById('total-time'),
   352→        currentStreak: document.getElementById('current-streak'),
   353→        taskInput: document.getElementById('task-input'),
   354→        addTaskBtn: document.getElementById('add-task-btn'),
   355→        activeTask: document.getElementById('active-task'),
   356→        activeTaskText: document.getElementById('active-task-text'),
   357→        removeTaskBtn: document.getElementById('remove-task-btn'),
   358→        settingsToggle: document.getElementById('settings-toggle'),
   359→        settingsPanel: document.getElementById('settings-panel'),
   360→        timerProgress: document.querySelector('.timer-progress'),
   361→        modeTabs: document.querySelectorAll('.mode-tab'),
   362→        durationBtns: document.querySelectorAll('.duration-btn'),
   363→        workDuration: document.getElementById('work-duration'),
   364→        breakDuration: document.getElementById('break-duration'),
   365→        longBreakDuration: document.getElementById('long-break-duration'),
   366→        sessionsCount: document.getElementById('sessions-count'),
   367→        autoStartBreaksToggle: document.getElementById('auto-start-breaks'),
   368→        autoStartWorkToggle: document.getElementById('auto-start-work'),
   369→        soundEnabledToggle: document.getElementById('sound-enabled')
   370→      };
   371→      
   372→      this.loadState();
   373→      this.bindEvents();
   374→      this.updateDisplay();
   375→      this.updateSessionDots();
   376→      this.updateStats();
   377→    }
   378→    
   379→    bindEvents() {
   380→      // Main controls
   381→      this.elements.startBtn.addEventListener('click', () => this.toggle());
   382→      this.elements.resetBtn.addEventListener('click', () => this.reset());
   383→      this.elements.skipBtn.addEventListener('click', () => this.skip());
   384→      
   385→      // Mode tabs
   386→      this.elements.modeTabs.forEach(tab => {
   387→        tab.addEventListener('click', (e) => {
   388→          const mode = e.currentTarget.dataset.mode;
   389→          this.setMode(mode);
   390→        });
   391→      });
   392→      
   393→      // Task input
   394→      this.elements.addTaskBtn.addEventListener('click', () => this.addTask());
   395→      this.elements.taskInput.addEventListener('keypress', (e) => {
   396→        if (e.key === 'Enter') this.addTask();
   397→      });
   398→      this.elements.removeTaskBtn.addEventListener('click', () => this.removeTask());
   399→      
   400→      // Settings
   401→      this.elements.settingsToggle.addEventListener('click', () => this.toggleSettings());
   402→      this.elements.durationBtns.forEach(btn => {
   403→        btn.addEventListener('click', (e) => this.changeDuration(e.currentTarget));
   404→      });
   405→      
   406→      // Toggles
   407→      this.elements.autoStartBreaksToggle.addEventListener('change', (e) => {
   408→        this.autoStartBreaks = e.target.checked;
   409→        this.saveState();
   410→      });
   411→      this.elements.autoStartWorkToggle.addEventListener('change', (e) => {
   412→        this.autoStartWork = e.target.checked;
   413→        this.saveState();
   414→      });
   415→      this.elements.soundEnabledToggle.addEventListener('change', (e) => {
   416→        this.soundEnabled = e.target.checked;
   417→        this.saveState();
   418→      });
   419→      
   420→      // Keyboard shortcuts
   421→      document.addEventListener('keydown', (e) => {
   422→        // Don't trigger if typing in input
   423→        if (e.target.tagName === 'INPUT') return;
   424→        
   425→        switch(e.code) {
   426→          case 'Space':
   427→            e.preventDefault();
   428→            this.toggle();
   429→            break;
   430→          case 'KeyR':
   431→            this.reset();
   432→            break;
   433→          case 'KeyS':
   434→            this.skip();
   435→            break;
   436→        }
   437→      });
   438→      
   439→      // Visibility change - pause when tab hidden (optional)
   440→      document.addEventListener('visibilitychange', () => {
   441→        if (document.hidden && this.isRunning) {
   442→          // Keep running in background, just update title
   443→          this.updatePageTitle();
   444→        }
   445→      });
   446→    }
   447→    
   448→    toggle() {
   449→      if (this.isRunning) {
   450→        this.pause();
   451→      } else {
   452→        this.start();
   453→      }
   454→    }
   455→    
   456→    start() {
   457→      this.isRunning = true;
   458→      this.updatePlayButton();
   459→      
   460→      this.intervalId = setInterval(() => {
   461→        this.tick();
   462→      }, 1000);
   463→      
   464→      // Request wake lock on mobile
   465→      this.requestWakeLock();
   466→    }
   467→    
   468→    pause() {
   469→      this.isRunning = false;
   470→      this.updatePlayButton();
   471→      
   472→      if (this.intervalId) {
   473→        clearInterval(this.intervalId);
   474→        this.intervalId = null;
   475→      }
   476→      
   477→      this.releaseWakeLock();
   478→    }
   479→    
   480→    reset() {
   481→      this.pause();
   482→      this.timeRemaining = this.durations[this.mode];
   483→      this.updateDisplay();
   484→      this.saveState();
   485→    }
   486→    
   487→    skip() {
   488→      this.completeSession(true);
   489→    }
   490→    
   491→    tick() {
   492→      if (this.timeRemaining > 0) {
   493→        this.timeRemaining--;
   494→        this.updateDisplay();
   495→        this.updatePageTitle();
   496→        this.saveState();
   497→      } else {
   498→        this.completeSession(false);
   499→      }
   500→    }
   501→    
   502→    completeSession(skipped = false) {
   503→      this.pause();
   504→      this.playSound();
   505→      
   506→      if (this.mode === 'work' && !skipped) {
   507→        this.completedPomodoros++;
   508→        this.totalFocusTime += Math.round(this.durations.work / 60);
   509→        this.currentStreak++;
   510→        this.lastCompletedDate = new Date().toDateString();
   511→        
   512→        // Update session dot
   513→        const dot = this.elements.sessionDots?.querySelector(`[data-session="${this.currentSession}"]`);
   514→        if (dot) dot.classList.add('completed');
   515→        
   516→        this.currentSession++;
   517→        
   518→        if (this.currentSession > this.sessionsUntilLong) {
   519→          this.currentSession = 1;
   520→          this.setMode('longBreak');
   521→          // Reset all dots
   522→          this.updateSessionDots();
   523→        } else {
   524→          this.setMode('break');
   525→        }
   526→        
   527→        if (this.autoStartBreaks && !skipped) {
   528→          setTimeout(() => this.start(), 1000);
   529→        }
   530→      } else {
   531→        this.setMode('work');
   532→        if (this.autoStartWork && !skipped) {
   533→          setTimeout(() => this.start(), 1000);
   534→        }
   535→      }
   536→      
   537→      this.updateStats();
   538→      this.showNotification();
   539→      this.saveState();
   540→    }
   541→    
   542→    setMode(mode) {
   543→      this.mode = mode;
   544→      this.timeRemaining = this.durations[mode];
   545→      this.isRunning = false;
   546→      
   547→      // Update tabs
   548→      this.elements.modeTabs.forEach(tab => {
   549→        tab.classList.toggle('active', tab.dataset.mode === mode);
   550→      });
   551→      
   552→      // Update mode label with i18n
   553→      const modeLabels = {
   554→        work: this.getI18n('pomodoro.mode.work') || 'Focus',
   555→        break: this.getI18n('pomodoro.mode.break') || 'Break',
   556→        longBreak: this.getI18n('pomodoro.mode.longBreak') || 'Long Break'
   557→      };
   558→      if (this.elements.modeLabel) {
   559→        this.elements.modeLabel.textContent = modeLabels[mode];
   560→      }
   561→      
   562→      // Update progress color
   563→      if (this.elements.timerProgress) {
   564→        this.elements.timerProgress.style.stroke = this.getModeColor(mode);
   565→      }
   566→      
   567→      this.updateDisplay();
   568→      this.updatePlayButton();
   569→      this.saveState();
   570→    }

   572→    getModeColor(mode) {
   573→      const colors = {
   574→        work: '#E53935',
   575→        break: '#25D366',
   576→        longBreak: '#5C6BC0'
   577→      };
   578→      return colors[mode] || colors.work;
   579→    }

   581→    updateDisplay() {
   582→      const minutes = Math.floor(this.timeRemaining / 60);
   583→      const seconds = this.timeRemaining % 60;
   584→      const timeString = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
   585→      
   586→      if (this.elements.timeDisplay) {
   587→        this.elements.timeDisplay.textContent = timeString;
   588→      }
   589→      
   590→      // Update circular progress
   591→      if (this.elements.timerProgress) {
   592→        const totalTime = this.durations[this.mode];
   593→        const progress = this.timeRemaining / totalTime;
   594→        const circumference = 2 * Math.PI * 120; // r=120
   595→        const offset = circumference * (1 - progress);
   596→        this.elements.timerProgress.style.strokeDashoffset = offset;
   597→      }
   598→      
   599→      // Update page title
   600→      this.updatePageTitle();
   601→    }

   604→      const minutes = Math.floor(this.timeRemaining / 60);
   605→      const seconds = this.timeRemaining % 60;
   606→      const prefix = this.isRunning ? '⏱' : '🍅';
   607→      document.title = `${prefix} ${minutes}:${seconds.toString().padStart(2, '0')} — OPN Chat`;
   608→    }

   616→        pauseIcon.style.display = this.isRunning ? 'block' : 'none';

   620→    updateSessionDots() {
   621→      if (!this.elements.sessionDots) return;

   624→      dots.forEach((dot, index) => {

   629→    updateStats() {
   630→      if (this.elements.completedCount) {
   631→        this.elements.completedCount.textContent = this.completedPomodoros;
   632→      }
   633→      if (this.elements.totalTime) {
   634→        this.elements.totalTime.textContent = `${this.totalFocusTime}m`;
   635→      }
   636→      if (this.elements.currentStreak) {
   637→        // Check if streak should reset (new day)

   639→        if (this.lastCompletedDate && this.lastCompletedDate !== today && this.completedPomodoros > 0) {
   640→          // Streak continues until next day without completion
   641→        }
   642→        this.elements.currentStreak.textContent = `🔥 ${this.currentStreak}`;
   643→      }
   644→    }

   646→    addTask() {
   647→      const text = this.elements.taskInput.value.trim();

   674→      this.elements.settingsPanel.style.display = isVisible ? 'none' : 'block';

   677→    changeDuration(btn) {
   678→      const setting = btn.dataset.setting;
   679→      const change = parseInt(btn.dataset.change);
   680→      
   681→      switch(setting) {
   682→        case 'work':
   683→          this.durations.work = Math.max(60, Math.max(1, this.durations.work / 60 + change)) * 60;
   684→          if (this.elements.workDuration) this.elements.workDuration.textContent = this.durations.work / 60;
   685→          break;
   686→        case 'break':
   687→          this.durations.break = Math.max(60, Math.max(1, this.durations.break / 60 + change)) * 60;
   688→          if (this.elements.breakDuration) this.elements.breakDuration.textContent = this.durations.break / 60;
   689→          break;
   690→        case 'longBreak':
   691→          this.durations.longBreak = Math.max(60, Math.max(1, this.durations.longBreak / 60 + change)) * 60;
   692→          if (this.elements.longBreakDuration) this.elements.longBreakDuration.textContent = this.durations.longBreak / 60;
   693→          break;
   694→        case 'sessions':
   695→          this.sessionsUntilLong = Math.max(2, Math.min(10, this.sessionsUntilLong + change));
   696→          if (this.elements.sessionsCount) this.elements.sessionsCount.textContent = this.sessionsUntilLong;
   697→          this.updateSessionDots();
   698→          break;
   699→      }
   700→      
   701→      if (this.mode === setting || (setting === 'sessions')) {
   702→        this.timeRemaining = this.durations[this.mode];
   703→        this.updateDisplay();
   704→      }
   705→      
   706→      this.saveState();
   707→    }

   709→    // ===== AUDIO =====
   710→    playSound() {
   711→      if (!this.soundEnabled) return;
   712→      
   713→      try {
   714→        if (!this.audioContext) {
   715→          this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
   716→        }
   717→        
   718→        const oscillator = this.audioContext.createOscillator();
   719→        const gainNode = this.audioContext.createGain();
   720→        
   721→        oscillator.connect(gainNode);
   722→        gainNode.connect(this.audioContext.destination);
   723→        
   724→        oscillator.frequency.setValueAtTime(800, this.audioContext.currentTime);
   725→        oscillator.type = 'sine';
   726→        
   727→        gainNode.gain.setValueAtTime(0.3, this.audioContext.currentTime);
   728→        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.5);
   729→        
   730→        oscillator.start(this.audioContext.currentTime);
   731→        oscillator.stop(this.audioContext.currentTime + 0.5);
   732→      } catch (e) {
   733→        console.log('Audio not available:', e);
   734→      }
   735→    }

   787→    // ===== STATE PERSISTENCE =====
   788→    saveState() {
   789→      const state = {
   790→        mode: this.mode,
   791→        timeRemaining: this.timeRemaining,
   792→        isRunning: this.isRunning,
   793→        currentSession: this.currentSession,
   794→        completedPomodoros: this.completedPomodoros,
   795→        totalFocusTime: this.totalFocusTime,
   796→        currentStreak: this.currentStreak,
   797→        lastCompletedDate: this.lastCompletedDate,
   798→        durations: this.durations,
   799→        sessionsUntilLong: this.sessionsUntilLong,
   800→        autoStartBreaks: this.autoStartBreaks,
   801→        autoStartWork: this.autoStartWork,
   802→        soundEnabled: this.soundEnabled,
   803→        taskText: this.elements.activeTaskText?.textContent || ''
   804→      };
   805→      
   806→      try {
   807→        localStorage.setItem('pomodoro-state', JSON.stringify(state));
   808→      } catch (e) {
   809→        console.log('Could not save state:', e);
   810→      }
   811→    }
   812→    
   813→    loadState() {
   814→      try {
   815→        const saved = localStorage.getItem('pomodoro-state');
   816→        if (saved) {
   817→          const state = JSON.parse(saved);
   818→          
   819→          // Only restore if saved today
   820→          if (state.lastCompletedDate) {
   821→            const savedDate = new Date(state.lastCompletedDate);
   822→            const today = new Date();
   823→            if (savedDate.toDateString() !== today.toDateString()) {
   824→              // New day - reset daily stats but keep settings
   825→              this.durations = state.durations || this.durations;
   826→              this.sessionsUntilLong = state.sessionsUntilLong || this.sessionsUntilLong;
   827→              this.autoStartBreaks = state.autoStartBreaks ?? this.autoStartBreaks;
   828→              this.autoStartWork = state.autoStartWork ?? this.autoStartWork;
   829→              this.soundEnabled = state.soundEnabled ?? this.soundEnabled;
   830→              return;
   831→            }
   832→          }
   833→          
   834→          Object.assign(this, state);
   835→          
   836→          // Restore UI after a short delay for DOM
   837→          setTimeout(() => {
   838→            this.updateDisplay();
   839→            this.updateSessionDots();
   840→            this.updateStats();
   841→            this.updatePlayButton();
   842→            
   843→            // Restore task
   844→            if (state.taskText && this.elements.activeTaskText) {
   845→              this.elements.activeTaskText.textContent = state.taskText;
   846→              if (this.elements.activeTask) {
   847→                this.elements.activeTask.style.display = 'flex';
   848→              }
   849→            }
   850→            
   851→            // Restore settings UI
   852→            if (this.elements.workDuration) this.elements.workDuration.textContent = this.durations.work / 60;
   853→            if (this.elements.breakDuration) this.elements.breakDuration.textContent = this.durations.break / 60;
   854→            if (this.elements.longBreakDuration) this.elements.longBreakDuration.textContent = this.durations.longBreak / 60;
   855→            if (this.elements.sessionsCount) this.elements.sessionsCount.textContent = this.sessionsUntilLong;
   856→            if (this.elements.autoStartBreaksToggle) this.elements.autoStartBreaksToggle.checked = this.autoStartBreaks;
   857→            if (this.elements.autoStartWorkToggle) this.elements.autoStartWorkToggle.checked = this.autoStartWork;
   858→            if (this.elements.soundEnabledToggle) this.elements.soundEnabledToggle.checked = this.soundEnabled;
   859→            
   860→            // Update mode tabs
   861→            this.elements.modeTabs.forEach(tab => {
   862→              tab.classList.toggle('active', tab.dataset.mode === this.mode);
   863→            });
   864→            
   865→            // Update progress color
   866→            if (this.elements.timerProgress) {
   867→              this.elements.timerProgress.style.stroke = this.getModeColor(this.mode);
   868→            }
   869→            
   870→            // Resume if was running
   871→            if (this.isRunning) {
   872→              this.start();
   873→            }
   874→          }, 100);
   875→        }
   876→      } catch (e) {
   877→        console.log('Could not load state:', e);
   878→      }
   879→      
   880→      // Request notification permission on first interaction
   881→      document.addEventListener('click', () => this.requestNotificationPermission(), { once: true });
   882→    }
   883→    
   884→    // Helper to get i18n strings from data attributes
   885→    getI18n(key) {
   886→      const el = document.querySelector(`[data-i18n="${key}"]`);
   887→      return el?.textContent || null;
   888→    }
