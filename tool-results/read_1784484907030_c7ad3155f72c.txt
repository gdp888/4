   927→<script define:vars={{ lang }}>
   928→  const form = document.getElementById('whatsapp-form');
   929→  const phoneInput = document.getElementById('phone');
   930→  const messageInput = document.getElementById('message');
   931→  const errorMsg = document.getElementById('error-msg');
   932→  const historySection = document.getElementById('history-section');
   933→  const historyList = document.getElementById('history-list');
   934→  const clearBtn = document.getElementById('clear-history');
   935→
   936→  const STORAGE_KEY = 'opnchat-history';
   937→  const MAX_HISTORY = 5;
   938→
   939→  const localeMap = { en: 'en-US', es: 'es-ES', pt: 'pt-BR', id: 'id-ID' };
   940→  const locale = localeMap[lang] || 'en-US';
   941→
   942→  const chatLabel = lang === 'es' ? 'Chat' : lang === 'pt' ? 'Chat' : 'Chat';
   943→
   944→  // Phone input — digits only
   945→  phoneInput.addEventListener('input', () => {
   946→    let val = phoneInput.value.replace(/\D/g, '');
   947→    if (val.length > 15) val = val.slice(0, 15);
   948→    phoneInput.value = val;
   949→    errorMsg.textContent = '';
   950→  });
   951→
   952→  // Load history from localStorage
   953→  function loadHistory() {
   954→    try {
   955→      const data = localStorage.getItem(STORAGE_KEY);
   956→      return data ? JSON.parse(data) : [];
   957→    } catch {
   958→      return [];
   959→    }
   960→  }
   961→
   962→  // Save — store clean digits (no +), display with +
   963→  function saveHistory(cleanDigits) {
   964→    const history = loadHistory().filter(h => h.number !== cleanDigits);
   965→    history.unshift({ number: cleanDigits, timestamp: Date.now() });
   966→    if (history.length > MAX_HISTORY) history.pop();
   967→    localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
   968→    renderHistory();
   969→  }
   970→
   971→  function renderHistory() {
   972→    const history = loadHistory();
   973→    if (history.length === 0) {
   974→      historySection.style.display = 'none';
   975→      return;
   976→    }
   977→    historySection.style.display = 'block';
   978→    historyList.innerHTML = '';
   979→
   980→    history.forEach(item => {
   981→      const digits = item.number.replace(/\D/g, '');
   982→      const li = document.createElement('li');
   983→      li.className = 'history-item';
   984→
   985→      // Left side: number + time
   986→      const leftDiv = document.createElement('div');
   987→      leftDiv.className = 'history-left';
   988→
   989→      const numSpan = document.createElement('span');
   990→      numSpan.className = 'history-number';
   991→      numSpan.textContent = '+' + digits;
   992→      // Click number → fill input
   993→      numSpan.addEventListener('click', (e) => {
   994→        e.stopPropagation();
   995→        phoneInput.value = digits;
   996→        phoneInput.focus();
   997→      });
   998→
   999→      const timeSpan = document.createElement('span');
  1000→      timeSpan.className = 'history-time';
  1001→      const time = new Date(item.timestamp);
  1002→      timeSpan.textContent = time.toLocaleTimeString(locale, { hour: '2-digit', minute: '2-digit' });
  1003→
  1004→      leftDiv.appendChild(numSpan);
  1005→      leftDiv.appendChild(timeSpan);
  1006→
  1007→      // Right side: Chat button
  1008→      const chatBtn = document.createElement('button');
  1009→      chatBtn.type = 'button';
  1010→      chatBtn.className = 'history-chat-btn';
  1011→      chatBtn.textContent = chatLabel;
  1012→      chatBtn.addEventListener('click', (e) => {
  1013→        e.stopPropagation();
  1014→        window.open(`https://wa.me/${digits}`, '_blank', 'noopener,noreferrer');
  1015→      });
  1016→
  1017→      li.appendChild(leftDiv);
  1018→      li.appendChild(chatBtn);
  1019→      historyList.appendChild(li);
  1020→    });
  1021→  }
  1022→
  1023→  // Clear history
  1024→  clearBtn.addEventListener('click', () => {
  1025→    localStorage.removeItem(STORAGE_KEY);
  1026→    renderHistory();
  1027→  });
  1028→
  1029→  // Initial render
  1030→  renderHistory();
  1031→
  1032→  // Form submit
  1033→  form.addEventListener('submit', (e) => {
  1034→    e.preventDefault();
  1035→    errorMsg.textContent = '';
  1036→
  1037→    const rawPhone = phoneInput.value.replace(/\D/g, '');
  1038→    const message = messageInput.value.trim();
  1039→
  1040→    const errorMessages = {
  1041→      valid: lang === 'es' ? 'Ingresa un número de teléfono válido' : lang === 'pt' ? 'Insira um número de telefone válido' : 'Please enter a valid phone number',
  1042→      short: lang === 'es' ? 'El número de teléfono es demasiado corto' : lang === 'pt' ? 'O número de telefone é muito curto' : 'Phone number is too short',
  1043→    };
  1044→
  1045→    if (rawPhone.length < 7) {
  1046→      errorMsg.textContent = errorMessages.valid;
  1047→      phoneInput.focus();
  1048→      return;
  1049→    }
  1050→
  1051→    if (rawPhone.length < 10) {
  1052→      errorMsg.textContent = errorMessages.short;
  1053→      phoneInput.focus();
  1054→      return;
  1055→    }
  1056→
  1057→    // Save clean digits (without +)
  1058→    saveHistory(rawPhone);
  1059→
  1060→    let url = `https://wa.me/${rawPhone}`;
  1061→    if (message) {
  1062→      url += `?text=${encodeURIComponent(message)}`;
  1063→    }
  1064→    window.open(url, '_blank');
  1065→  });
  1066→
  1067→  // Keyboard shortcut — Enter to submit when phone is focused
  1068→  phoneInput.addEventListener('keydown', (e) => {
  1069→    if (e.key === 'Enter') {
  1070→      e.preventDefault();
  1071→      form.dispatchEvent(new Event('submit', { cancelable: true }));
  1072→    }
  1073→  });
  1074→
  1075→  // Theme toggle
  1076→  const themeToggle = document.getElementById('theme-toggle');
  1077→  if (themeToggle) {
  1078→    themeToggle.addEventListener('click', () => {
  1079→      const current = document.documentElement.getAttribute('data-theme');
  1080→      const next = current === 'dark' ? 'light' : 'dark';
  1081→      document.documentElement.setAttribute('data-theme', next);
  1082→      localStorage.setItem('opnchat-theme', next);
  1083→    });
  1084→  }
