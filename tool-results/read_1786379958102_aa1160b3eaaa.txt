   372→   372→          new QRCode(container, {

   386→   386→    // Generate link
   387→   387→    function doGenerate() {
   388→   388→      var username = usernameInput.value.trim().replace(/^@/, '');
   389→   389→      var message = messageInput.value.trim();
   390→   390→
   391→   391→      if (!username) {
   392→   392→        errorMsg.textContent = usernameInput.getAttribute('data-error-empty');
   393→   393→        errorMsg.style.display = 'block';
   394→   394→        return;
   395→   395→      }
   396→   396→
   397→   397→      // Validate username: 5-32 chars, letters, digits, underscores
   398→   398→      if (!/^[a-zA-Z][a-zA-Z0-9_]{4,31}$/.test(username)) {
   399→   399→        errorMsg.textContent = usernameInput.getAttribute('data-error-format');
   400→   400→        errorMsg.style.display = 'block';
   401→   401→        return;
   402→   402→      }
   403→   403→
   404→   404→      errorMsg.style.display = 'none';
   405→   405→
   406→   406→      var link = 'https://t.me/' + username;
   407→   407→      if (message) {
   408→   408→        if (selectedLinkType === 'bot') {
   409→   409→          link += '?start=' + encodeURIComponent(message);
   410→   410→        } else {
   411→   411→          link += '?text=' + encodeURIComponent(message);
   412→   412→        }
   413→   413→      }
   414→   414→
   415→   415→      generatedLink = link;
   416→   416→      linkDisplay.textContent = link;
   417→   417→      resultSection.style.display = 'block';
   418→   418→      previewMsg.textContent = message || '👋';
   419→   419→      previewMsg.style.display = 'block';
   420→   420→
   421→   421→      var qrCanvas = document.getElementById('tgl-qr-canvas');
   422→   422→      generateQR(link, qrCanvas);
   423→   423→
   424→   424→      resultSection.scrollIntoView({ behavior: 'smooth', block: 'start' });
   425→   425→    }

   435→   435→        var textarea = document.createElement('textarea');
   436→   436→        textarea.value = text;
   437→   437→        document.body.appendChild(textarea);
   438→   438→        textarea.select();
   439→   439→        document.execCommand('copy');
   440→   440→        document.body.removeChild(textarea);
   441→   441→      });
   442→   442→    }
