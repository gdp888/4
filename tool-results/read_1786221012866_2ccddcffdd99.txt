   263→   263→  const timeInput = document.getElementById('time');

   265→   265→  const submitBtn = document.getElementById('submit-btn');
   266→   266→  const successMessage = document.getElementById('success-message');

   268→   268→  const taskLink = document.getElementById('task-link');

   270→   270→  const scheduleAnotherBtn = document.getElementById('schedule-another');
   271→   271→  const tasksList = document.getElementById('tasks-list');
   272→   272→  const timezoneDisplay = document.getElementById('timezone-display');
   273→   273→
   274→   274→  // Set minimum date to today
   275→   275→  const today = new Date().toISOString().split('T')[0];
   276→   276→  dateInput.setAttribute('min', today);
   277→   277→
   278→   278→  // Display user's timezone
   279→   279→  const userTimezone = Intl.DateTimeFormat().resolvedOptions().timeZone;
   280→   280→  timezoneDisplay.textContent = `Timezone: ${userTimezone}`;
   281→   281→
   282→   282→  // Character counter
   283→   283→  messageInput.addEventListener('input', () => {
   284→   284→    charCount.textContent = messageInput.value.length;
   285→   285→  });
   286→   286→
   287→   287→  // Format phone number as user types
   288→   288→  phoneInput.addEventListener('input', (e) => {
   289→   289→    let value = e.target.value.replace(/\D/g, '');
   290→   290→    if (value.length > 15) value = value.slice(0, 15);
   291→   291→    e.target.value = value;
   292→   292→  });
   293→   293→
   294→   294→  // Form submission
   295→   295→  form.addEventListener('submit', async (e) => {
   296→   296→    e.preventDefault();
   297→   297→    
   298→   298→    const phone = phoneInput.value.replace(/\D/g, '');
   299→   299→    const message = messageInput.value.trim();
   300→   300→    const date = dateInput.value;
   301→   301→    const time = timeInput.value;
   302→   302→
   303→   303→    // Validate
   304→   304→    if (phone.length < 8 || phone.length > 15) {
   305→   305→      alert('Please enter a valid phone number');
   306→   306→      return;
   307→   307→    }
   308→   308→
   309→   309→    const scheduledDateTime = new Date(`${date}T${time}`);
   310→   310→    if (scheduledDateTime <= new Date()) {
   311→   311→      alert('Please select a future date and time');
   312→   312→      return;
   313→   313→    }
   314→   314→
   315→   315→    // Show loading state
   316→   316→    submitBtn.disabled = true;
   317→   317→    submitBtn.innerHTML = '<span class="spinner"></span> Scheduling...';
   318→   318→
   319→   319→    try {
   320→   320→      const response = await fetch('/api/scheduler', {
   321→   321→        method: 'POST',
   322→   322→        headers: { 'Content-Type': 'application/json' },
   323→   323→        body: JSON.stringify({
   324→   324→          phone,
   325→   325→          message,
   326→   326→          scheduledFor: scheduledDateTime.toISOString(),
   327→   327→          timezone: userTimezone,
   328→   328→          lang
   329→   329→        })
   330→   330→      });
   331→   331→
   332→   332→      const data = await response.json();
   333→   333→
   334→   334→      if (data.success) {
   335→   335→        // Show success
   336→   336→        form.classList.add('hidden');
   337→   337→        successMessage.classList.remove('hidden');
   338→   338→        
   339→   339→        const formattedDate = new Date(data.task.scheduledFor).toLocaleString(lang);
   340→   340→        successDetails.textContent = `${data.task.phone} at ${formattedDate}`;
   341→   341→        
   342→   342→        taskLink.href = `/scheduler/task/${data.task.id}`;
   343→   343→        taskLink.textContent = `${window.location.origin}/scheduler/task/${data.task.id}`;
   344→   344→        
   345→   345→        // Save task ID to localStorage for listing
   346→   346→        saveTaskLocally(data.task);
   347→   347→        
   348→   348→        // Refresh tasks list
   349→   349→        loadTasks();
   350→   350→      } else {
   351→   351→        alert(data.error || 'Failed to schedule message');
   352→   352→      }
   353→   353→    } catch (error) {
   354→   354→      console.error('Error:', error);
   355→   355→      alert('Network error. Please try again.');
   356→   356→    }
   357→   357→
   358→   358→    submitBtn.disabled = false;

   388→   388→  // Local storage functions for tasks
   389→   389→  function saveTaskLocally(task) {
   390→   390→    const tasks = JSON.parse(localStorage.getItem('opnchat-scheduled-tasks') || '[]');
   391→   391→    tasks.unshift(task);
   392→   392→    // Keep only last 50 tasks
   393→   393→    if (tasks.length > 50) tasks.pop();
   394→   394→    localStorage.setItem('opnchat-scheduled-tasks', JSON.stringify(tasks));
   395→   395→  }
   396→   396→
   397→   397→  function loadTasks() {
   398→   398→    const tasks = JSON.parse(localStorage.getItem('opnchat-scheduled-tasks') || '[]');
   399→   399→    
   400→   400→    if (tasks.length === 0) {
   401→   401→      tasksList.innerHTML = `
   402→   402→        <div class="empty-state">
   403→   403→          <svg class="empty-icon" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5">
   404→   404→            <rect x="3" y="4" width="18" height="18" rx="2"/>
   405→   405→            <line x1="16" y1="2" x2="16" y2="6"/>
   406→   406→            <line x1="8" y1="2" x2="8" y2="6"/>
   407→   407→            <line x1="3" y1="10" x2="21" y2="10"/>
   408→   408→            <line x1="9" y1="16" x2="15" y2="16"/>
   409→   409→          </svg>
   410→   410→          <p>No scheduled messages yet</p>
   411→   411→        </div>
   412→   412→      `;
   413→   413→      return;
   414→   414→    }
   415→   415→
   416→   416→    tasksList.innerHTML = tasks.map(task => {
   417→   417→      const scheduledDate = new Date(task.scheduledFor);
   418→   418→      const now = new Date();
   419→   419→      const isPast = scheduledDate <= now;
   420→   420→      const statusClass = isPast ? 'status-past' : 'status-pending';
   421→   421→      const statusText = isPast ? 'Ready to send' : 'Scheduled';
   422→   422→      
   423→   423→      return `
   424→   424→        <div class="task-item ${statusClass}">
   425→   425→          <div class="task-info">
   426→   426→            <span class="task-phone">${task.phone.replace(/(\d{1,3})(\d{3})(\d{3})(\d{2,})/, '+$1 $2 $3 $4')}</span>
   427→   427→            <span class="task-time">${scheduledDate.toLocaleString(lang)}</span>
   428→   428→          </div>
   429→   429→          <div class="task-status">${statusText}</div>
   430→   430→          <a href="/scheduler/task/${task.id}" class="task-view-btn" target="_blank">View</a>
   431→   431→        </div>
   432→   432→      `;
   433→   433→    }).join('');
   434→   434→  }
