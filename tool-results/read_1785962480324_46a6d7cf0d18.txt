   848→<script define:vars={{ lang }}>
   849→  // Desktop detection — show download link if user is on desktop
   850→  const isDesktop = !/Android|iPhone|iPad|iPod|Opera Mini|IEMobile|WPDesktop/i.test(navigator.userAgent);
   851→  const hintDownloadLink = document.getElementById('viber-download-link');
   852→  const hintDownloadText = document.getElementById('viber-hint-text');
   853→  if (isDesktop && hintDownloadLink) {
   854→    hintDownloadLink.style.display = 'inline-flex';
   855→  }
   856→
   857→  const form = document.getElementById('viber-form');
   858→  const phoneInput = document.getElementById('phone');
   859→  const messageInput = document.getElementById('message');
   860→  const errorMsg = document.getElementById('error-msg');
   861→  const historySection = document.getElementById('history-section');
   862→  const historyList = document.getElementById('history-list');
   863→  const clearBtn = document.getElementById('clear-history');
   864→
   865→  const STORAGE_KEY = 'opnchat-viber-history';
   866→  const MAX_HISTORY = 5;
   867→
   868→  const localeMap = { en: 'en-US', es: 'es-ES', pt: 'pt-BR', id: 'id-ID', ru: 'ru-RU', ar: 'ar-SA', de: 'de-DE' };
   869→  const locale = localeMap[lang] || 'en-US';
   870→
   871→  const chatLabel = 'Viber';
   872→
   873→  // Phone input — digits only
   874→  phoneInput.addEventListener('input', () => {
   875→    let val = phoneInput.value.replace(/\D/g, '');
   876→    if (val.length > 15) val = val.slice(0, 15);
   877→    phoneInput.value = val;
   878→    errorMsg.textContent = '';
   879→  });
   880→
   881→  // Load history from localStorage
   882→  function loadHistory() {
   883→    try {
   884→      const data = localStorage.getItem(STORAGE_KEY);
   885→      return data ? JSON.parse(data) : [];
   886→    } catch {
   887→      return [];
   888→    }
   889→  }
   890→
   891→  // Save — store clean digits (no +), display with +
   892→  function saveHistory(cleanDigits) {
   893→    const history = loadHistory().filter(h => h.number !== cleanDigits);
   894→    history.unshift({ number: cleanDigits, timestamp: Date.now() });
   895→    if (history.length > MAX_HISTORY) history.pop();
   896→    localStorage.setItem(STORAGE_KEY, JSON.stringify(history));
   897→    renderHistory();
   898→  }
   899→
   900→  function renderHistory() {
   901→    const history = loadHistory();
   902→    if (history.length === 0) {
   903→      historySection.style.display = 'none';
   904→      return;
   905→    }
   906→    historySection.style.display = 'block';
   907→    historyList.innerHTML = '';
   908→
   909→    history.forEach(item => {
   910→      const digits = item.number.replace(/\D/g, '');
   911→      const li = document.createElement('li');
   912→      li.className = 'history-item';
   913→
   914→      // Left side: number + time
   915→      const leftDiv = document.createElement('div');
   916→      leftDiv.className = 'history-left';
   917→
   918→      const numSpan = document.createElement('span');
   919→      numSpan.className = 'history-number';
   920→      numSpan.textContent = '+' + digits;
   921→      // Click number → fill input
   922→      numSpan.addEventListener('click', (e) => {
   923→        e.stopPropagation();
   924→        phoneInput.value = digits;
   925→        phoneInput.focus();
   926→      });
   927→
   928→      const timeSpan = document.createElement('span');
   929→      timeSpan.className = 'history-time';
   930→      const time = new Date(item.timestamp);
   931→      timeSpan.textContent = time.toLocaleTimeString(locale, { hour: '2-digit', minute: '2-digit' });
   932→
   933→      leftDiv.appendChild(numSpan);
   934→      leftDiv.appendChild(timeSpan);
   935→
   936→      // Right side: Chat button
   937→      const chatBtn = document.createElement('button');
   938→      chatBtn.type = 'button';
   939→      chatBtn.className = 'history-chat-btn';
   940→      chatBtn.textContent = chatLabel;
   941→      chatBtn.addEventListener('click', (e) => {
   942→        e.stopPropagation();
   943→        window.open(`viber://chat?number=%2B${digits}`, '_self');
   944→      });
   945→
   946→      li.appendChild(leftDiv);
   947→      li.appendChild(chatBtn);
   948→      historyList.appendChild(li);
   949→    });
   950→  }
   951→
   952→  // Clear history
   953→  clearBtn.addEventListener('click', () => {
   954→    localStorage.removeItem(STORAGE_KEY);
   955→    renderHistory();
   956→  });
   957→
   958→  // Initial render
   959→  renderHistory();
   960→
   961→  // Form submit
   962→  form.addEventListener('submit', (e) => {
   963→    e.preventDefault();
   964→    errorMsg.textContent = '';
   965→
   966→    const rawPhone = phoneInput.value.replace(/\D/g, '');
   967→    const message = messageInput.value.trim();
   968→
   969→    // Error messages by language
   970→    const errorMessages = {
   971→      valid: lang === 'ru' ? 'Введите корректный номер телефона' : lang === 'es' ? 'Ingresa un número de teléfono válido' : lang === 'pt' ? 'Insira um número de telefone válido' : lang === 'de' ? 'Bitte geben Sie eine gültige Telefonnummer ein' : lang === 'id' ? 'Masukkan nomor telepon yang valid' : lang === 'ar' ? 'أدخل رقم هاتف صالح' : 'Please enter a valid phone number',
   972→      short: lang === 'ru' ? 'Номер телефона слишком короткий' : lang === 'es' ? 'El número de teléfono es demasiado corto' : lang === 'pt' ? 'O número de telefone é muito curto' : lang === 'de' ? 'Die Telefonnummer ist zu kurz' : lang === 'id' ? 'Nomor telepon terlalu pendek' : lang === 'ar' ? 'رقم الهاتف قصير جداً' : 'Phone number is too short',
   973→    };
   974→
   975→    if (rawPhone.length < 7) {
   976→      errorMsg.textContent = errorMessages.valid;
   977→      phoneInput.focus();
   978→      return;
   979→    }
   980→
   981→    if (rawPhone.length < 10) {
   982→      errorMsg.textContent = errorMessages.short;
   983→      phoneInput.focus();
   984→      return;
   985→    }
   986→
   987→    // Save clean digits (without +)
   988→    saveHistory(rawPhone);
   989→
   990→    // Viber deep link: viber://chat?number=%2B{phone}
   991→    // Note: Viber requires the + prefix (URL-encoded as %2B)
   992→    let url = `viber://chat?number=%2B${rawPhone}`;
   993→    if (message) {
   994→      url += `&text=${encodeURIComponent(message)}`;
   995→    }
   996→    window.open(url, '_self');
   997→  });
   998→
   999→  // Keyboard shortcut — Enter to submit when phone is focused
  1000→  phoneInput.addEventListener('keydown', (e) => {
  1001→    if (e.key === 'Enter') {
  1002→      e.preventDefault();
  1003→      form.dispatchEvent(new Event('submit', { cancelable: true }));
  1004→    }
  1005→  });
