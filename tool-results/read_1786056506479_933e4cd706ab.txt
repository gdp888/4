   826→  function parseOffsetToMinutes(offsetStr: string): number {

   830→    const hours = parseInt(match[2]) || 0;
   831→    const minutes = parseInt(match[3]) || 0;
   832→    return sign * (hours * 60 + minutes);

   885→      timeZone: tzString,
   886→      hour: 'numeric',

   888→      second: 'numeric',

   890→      year: 'numeric',

   894→    });

   896→    
   897→    const getVal = (type: Intl.DateTimeFormatPartTypes) => parts.find(p => p.type === type)?.value;
   898→    return new Date(
   899→      parseInt(getVal('year')!),
   900→      parseInt(getVal('month')!) - 1,

   902→      parseInt(getVal('hour')!),
