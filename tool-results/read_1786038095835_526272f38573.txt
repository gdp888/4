   414→   414→      });
   415→   415→      this.elements.soundEnabledToggle.addEventListener('change', (e) => {
   416→   416→        this.soundEnabled = e.target.checked;
   417→   417→        this.saveState();
   418→   418→      });
   419→   419→      
   420→   420→      // Keyboard shortcuts
   421→   421→      document.addEventListener('keydown', (e) => {
   422→   422→        // Don't trigger if typing in input
   423→   423→        if (e.target.tagName === 'INPUT') return;
   424→   424→        
   425→   425→        switch(e.code) {
   426→   426→          case 'Space':
   427→   427→            e.preventDefault();
   428→   428→            this.toggle();
   429→   429→            break;
   430→   430→          case 'KeyR':
   431→   431→            this.reset();
   432→   432→            break;
   433→   433→          case 'KeyS':
   434→   434→            this.skip();
   435→   435→            break;
   436→   436→        }
   437→   437→      });
   438→   438→      
   439→   439→      // Visibility change - pause when tab hidden (optional)
   440→   440→      document.addEventListener('visibilitychange', () => {
   441→   441→        if (document.hidden && this.isRunning) {
   442→   442→          // Keep running in background, just update title
   443→   443→          this.updatePageTitle();
   444→   444→        }
   445→   445→      });
   446→   446→    }
   447→   447→    
   448→   448→    toggle() {
   449→   449→      if (this.isRunning) {
   450→   450→        this.pause();
   451→   451→      } else {
   452→   452→        this.start();
   453→   453→      }
   454→   454→    }
   455→   455→    
   456→   456→    start() {
   457→   457→      this.isRunning = true;
   458→   458→      this.updatePlayButton();
   459→   459→      
   460→   460→      this.intervalId = setInterval(() => {
   461→   461→        this.tick();
   462→   462→      }, 1000);
   463→   463→      
   464→   464→      // Request wake lock on mobile
   465→   465→      this.requestWakeLock();
   466→   466→    }
   467→   467→    
   468→   468→    pause() {
   469→   469→      this.isRunning = false;
   470→   470→      this.updatePlayButton();
   471→   471→      
   472→   472→      if (this.intervalId) {
   473→   473→        clearInterval(this.intervalId);
   474→   474→        this.intervalId = null;
   475→   475→      }
   476→   476→      
   477→   477→      this.releaseWakeLock();
   478→   478→    }
   479→   479→    
   480→   480→    reset() {
   481→   481→      this.pause();
   482→   482→      this.timeRemaining = this.durations[this.mode];
   483→   483→      this.updateDisplay();
   484→   484→      this.saveState();
   485→   485→    }
   486→   486→    
   487→   487→    skip() {
   488→   488→      this.completeSession(true);
   489→   489→    }
   490→   490→    
   491→   491→    tick() {
   492→   492→      if (this.timeRemaining > 0) {
   493→   493→        this.timeRemaining--;
   494→   494→        this.updateDisplay();
   495→   495→        this.updatePageTitle();
   496→   496→        this.saveState();
   497→   497→      } else {
   498→   498→        this.completeSession(false);
   499→   499→      }
   500→   500→    }
   501→   501→    
   502→   502→    completeSession(skipped = false) {
   503→   503→      this.pause();
   504→   504→      this.playSound();
   505→   505→      
   506→   506→      if (this.mode === 'work' && !skipped) {
   507→   507→        this.completedPomodoros++;
   508→   508→        this.totalFocusTime += Math.round(this.durations.work / 60);
   509→   509→        this.currentStreak++;
   510→   510→        this.lastCompletedDate = new Date().toDateString();
   511→   511→        
   512→   512→        // Update session dot
   513→   513→        const dot = this.elements.sessionDots?.querySelector(`[data-session="${this.currentSession}"]`);
   514→   514→        if (dot) dot.classList.add('completed');
   515→   515→        
   516→   516→        this.currentSession++;
   517→   517→        
   518→   518→        if (this.currentSession > this.sessionsUntilLong) {
   519→   519→          this.currentSession = 1;
   520→   520→          this.setMode('longBreak');
   521→   521→          // Reset all dots
   522→   522→          this.updateSessionDots();
   523→   523→        } else {
   524→   524→          this.setMode('break');
   525→   525→        }
   526→   526→        
   527→   527→        if (this.autoStartBreaks && !skipped) {
   528→   528→          setTimeout(() => this.start(), 1000);
   529→   529→        }
   530→   530→      } else {
   531→   531→        this.setMode('work');
   532→   532→        if (this.autoStartWork && !skipped) {
   533→   533→          setTimeout(() => this.start(), 1000);
   534→   534→        }
   535→   535→      }
   536→   536→      
   537→   537→      this.updateStats();
   538→   538→      this.showNotification();
   539→   539→      this.saveState();
   540→   540→    }
   541→   541→    
   542→   542→    setMode(mode) {
   543→   543→      this.mode = mode;
   544→   544→      this.timeRemaining = this.durations[mode];
   545→   545→      this.isRunning = false;
   546→   546→      
   547→   547→      // Update tabs
   548→   548→      this.elements.modeTabs.forEach(tab => {
   549→   549→        tab.classList.toggle('active', tab.dataset.mode === mode);
   550→   550→      });
   551→   551→      
   552→   552→      // Update mode label with i18n
   553→   553→      const modeLabels = {
   554→   554→        work: this.getI18n('pomodoro.mode.work') || 'Focus',
   555→   555→        break: this.getI18n('pomodoro.mode.break') || 'Break',
   556→   556→        longBreak: this.getI18n('pomodoro.mode.longBreak') || 'Long Break'
   557→   557→      };
   558→   558→      if (this.elements.modeLabel) {
   559→   559→        this.elements.modeLabel.textContent = modeLabels[mode];
   560→   560→      }
   561→   561→      
   562→   562→      // Update progress color
   563→   563→      if (this.elements.timerProgress) {
   564→   564→        this.elements.timerProgress.style.stroke = this.getModeColor(mode);
   565→   565→      }
   566→   566→      
   567→   567→      this.updateDisplay();
   568→   568→      this.updatePlayButton();
   569→   569→      this.saveState();
   570→   570→    }
   571→   571→    
   572→   572→    getModeColor(mode) {
   573→   573→      const colors = {
   574→   574→        work: '#E53935',
   575→   575→        break: '#25D366',
   576→   576→        longBreak: '#5C6BC0'
   577→   577→      };
   578→   578→      return colors[mode] || colors.work;
   579→   579→    }
   580→   580→    
   581→   581→    updateDisplay() {
   582→   582→      const minutes = Math.floor(this.timeRemaining / 60);
   583→   583→      const seconds = this.timeRemaining % 60;
   584→   584→      const timeString = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;
   585→   585→      
   586→   586→      if (this.elements.timeDisplay) {
   587→   587→        this.elements.timeDisplay.textContent = timeString;
   588→   588→      }
   589→   589→      
   590→   590→      // Update circular progress
   591→   591→      if (this.elements.timerProgress) {
   592→   592→        const totalTime = this.durations[this.mode];
   593→   593→        const progress = this.timeRemaining / totalTime;
   594→   594→        const circumference = 2 * Math.PI * 120; // r=120
   595→   595→        const offset = circumference * (1 - progress);
   596→   596→        this.elements.timerProgress.style.strokeDashoffset = offset;
   597→   597→      }
   598→   598→      
   599→   599→      // Update page title
   600→   600→      this.updatePageTitle();
   601→   601→    }
   602→   602→    
   603→   603→    updatePageTitle() {
   604→   604→      const minutes = Math.floor(this.timeRemaining / 60);
   605→   605→      const seconds = this.timeRemaining % 60;
   606→   606→      const prefix = this.isRunning ? '⏱' : '🍅';
   607→   607→      document.title = `${prefix} ${minutes}:${seconds.toString().padStart(2, '0')} — OPN Chat`;
   608→   608→    }
   609→   609→    
   610→   610→    updatePlayButton() {
   611→   611→      const playIcon = this.elements.startBtn?.querySelector('.play-icon');
   612→   612→      const pauseIcon = this.elements.startBtn?.querySelector('.pause-icon');
   613→   613→      
   614→   614→      if (playIcon && pauseIcon) {
   615→   615→        playIcon.style.display = this.isRunning ? 'none' : 'block';
   616→   616→        pauseIcon.style.display = this.isRunning ? 'block' : 'none';
   617→   617→      }
   618→   618→    }
   619→   619→    
   620→   620→    updateSessionDots() {
   621→   621→      if (!this.elements.sessionDots) return;
   622→   622→      
   623→   623→      const dots = this.elements.sessionDots.querySelectorAll('.dot');
   624→   624→      dots.forEach((dot, index) => {
   625→   625→        dot.classList.toggle('completed', index < (this.currentSession - 1));
   626→   626→      });
   627→   627→    }
   628→   628→    
   629→   629→    updateStats() {
   630→   630→      if (this.elements.completedCount) {
   631→   631→        this.elements.completedCount.textContent = this.completedPomodoros;
   632→   632→      }
   633→   633→      if (this.elements.totalTime) {
   634→   634→        this.elements.totalTime.textContent = `${this.totalFocusTime}m`;
   635→   635→      }
   636→   636→      if (this.elements.currentStreak) {
   637→   637→        // Check if streak should reset (new day)
   638→   638→        const today = new Date().toDateString();
   639→   639→        if (this.lastCompletedDate && this.lastCompletedDate !== today && this.completedPomodoros > 0) {
   640→   640→          // Streak continues until next day without completion
   641→   641→        }
   642→   642→        this.elements.currentStreak.textContent = `🔥 ${this.currentStreak}`;
   643→   643→      }
   644→   644→    }
   645→   645→    
   646→   646→    addTask() {
   647→   647→      const text = this.elements.taskInput.value.trim();
   648→   648→      if (!text) return;
   649→   649→      
   650→   650→      if (this.elements.activeTaskText) {
   651→   651→        this.elements.activeTaskText.textContent = text;
   652→   652→      }
   653→   653→      if (this.elements.activeTask) {
   654→   654→        this.elements.activeTask.style.display = 'flex';
   655→   655→      }
   656→   656→      this.elements.taskInput.value = '';
   657→   657→      this.saveState();
   658→   658→    }
   659→   659→    
   660→   660→    removeTask() {
   661→   661→      if (this.elements.activeTaskText) {
   662→   662→        this.elements.activeTaskText.textContent = '';
   663→   663→      }
   664→   664→      if (this.elements.activeTask) {
   665→   665→        this.elements.activeTask.style.display = 'none';
   666→   666→      }
   667→   667→      this.saveState();
   668→   668→    }
   669→   669→    
   670→   670→    toggleSettings() {
   671→   671→      if (!this.elements.settingsPanel) return;
   672→   672→      
   673→   673→      const isVisible = this.elements.settingsPanel.style.display !== 'none';
   674→   674→      this.elements.settingsPanel.style.display = isVisible ? 'none' : 'block';
   675→   675→    }
   676→   676→    
   677→   677→    changeDuration(btn) {
   678→   678→      const setting = btn.dataset.setting;
   679→   679→      const change = parseInt(btn.dataset.change);
   680→   680→      
   681→   681→      switch(setting) {
   682→   682→        case 'work':
   683→   683→          this.durations.work = Math.max(60, Math.max(1, this.durations.work / 60 + change)) * 60;
   684→   684→          if (this.elements.workDuration) this.elements.workDuration.textContent = this.durations.work / 60;
   685→   685→          break;
   686→   686→        case 'break':
   687→   687→          this.durations.break = Math.max(60, Math.max(1, this.durations.break / 60 + change)) * 60;
   688→   688→          if (this.elements.breakDuration) this.elements.breakDuration.textContent = this.durations.break / 60;
   689→   689→          break;
   690→   690→        case 'longBreak':
   691→   691→          this.durations.longBreak = Math.max(60, Math.max(1, this.durations.longBreak / 60 + change)) * 60;
   692→   692→          if (this.elements.longBreakDuration) this.elements.longBreakDuration.textContent = this.durations.longBreak / 60;
   693→   693→          break;
   694→   694→        case 'sessions':
   695→   695→          this.sessionsUntilLong = Math.max(2, Math.min(10, this.sessionsUntilLong + change));
   696→   696→          if (this.elements.sessionsCount) this.elements.sessionsCount.textContent = this.sessionsUntilLong;
   697→   697→          this.updateSessionDots();
   698→   698→          break;
   699→   699→      }
   700→   700→      
   701→   701→      if (this.mode === setting || (setting === 'sessions')) {
   702→   702→        this.timeRemaining = this.durations[this.mode];
   703→   703→        this.updateDisplay();
   704→   704→      }
   705→   705→      
   706→   706→      this.saveState();
   707→   707→    }
   708→   708→    
   709→   709→    // ===== AUDIO =====
   710→   710→    playSound() {
   711→   711→      if (!this.soundEnabled) return;
   712→   712→      
   713→   713→      try {
   714→   714→        if (!this.audioContext) {
   715→   715→          this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
   716→   716→        }
   717→   717→        
   718→   718→        const oscillator = this.audioContext.createOscillator();
   719→   719→        const gainNode = this.audioContext.createGain();
   720→   720→        
   721→   721→        oscillator.connect(gainNode);
   722→   722→        gainNode.connect(this.audioContext.destination);
   723→   723→        
   724→   724→        oscillator.frequency.setValueAtTime(800, this.audioContext.currentTime);
   725→   725→        oscillator.type = 'sine';
   726→   726→        
   727→   727→        gainNode.gain.setValueAtTime(0.3, this.audioContext.currentTime);
   728→   728→        gainNode.gain.exponentialRampToValueAtTime(0.01, this.audioContext.currentTime + 0.5);
   729→   729→        
   730→   730→        oscillator.start(this.audioContext.currentTime);
   731→   731→        oscillator.stop(this.audioContext.currentTime + 0.5);
   732→   732→      } catch (e) {
   733→   733→        console.log('Audio not available:', e);
   734→   734→      }
   735→   735→    }
   736→   736→    
   737→   737→    showNotification() {
   738→   738→      if ('Notification' in window && Notification.permission === 'granted') {
   739→   739→        const messages = {
   740→   740→          work: this.getI18n('pomodoro.notification.workDone') || 'Time to take a break!',
   741→   741→          break: this.getI18n('pomodoro.notification.breakDone') || 'Time to focus!',
   742→   742→          longBreak: this.getI18n('pomodoro.notification.breakDone') || 'Time to focus!'
   743→   743→        };
   744→   744→        
   745→   745→        new Notification('🍅 OPN Chat', {
   746→   746→          body: messages[this.mode],
   747→   747→          icon: '/favicon.ico'
   748→   748→        });
   749→   749→      }
   750→   750→    }
   751→   751→    
   752→   752→    requestNotificationPermission() {
   753→   753→      if ('Notification' in window && Notification.permission === 'default') {
   754→   754→        Notification.requestPermission();
   755→   755→      }
   756→   756→    }

   758→   758→    // ===== WAKE LOCK (Mobile) =====
   759→   759→    async function getWakeLock() {
   760→   760→      try {
   761→   761→        if ('wakeLock' in navigator) {
   762→   762→          return await navigator.wakeLock.request('screen');
   763→   763→        }
   764→   764→      } catch (err) {
   765→   765→        console.log('Wake Lock error:', err);
   766→   766→      }
   767→   767→      return null;
   768→   768→    }
   769→   769→    
   770→   770→    let wakeLock = null;
   771→   771→    
   772→   772→    async requestWakeLock() {
   773→   773→      wakeLock = await getWakeLock();
   774→   774→    }
   775→   775→    
   776→   776→    releaseWakeLock() {
   777→   777→      if (wakeLock) {
   778→   778→        wakeLock.release();
   779→   779→        wakeLock = null;
   780→   780→      }
   781→   781→    }

   783→   783→    // Make methods available on instance
   784→   784→    this.requestWakeLock = requestWakeLock;
   785→   785→    this.releaseWakeLock = releaseWakeLock;
   786→   786→    
   787→   787→    // ===== STATE PERSISTENCE =====
   788→   788→    saveState() {
   789→   789→      const state = {
   790→   790→        mode: this.mode,
   791→   791→        timeRemaining: this.timeRemaining,
   792→   792→        isRunning: this.isRunning,
   793→   793→        currentSession: this.currentSession,
   794→   794→        completedPomodoros: this.completedPomodoros,
   795→   795→        totalFocusTime: this.totalFocusTime,
   796→   796→        currentStreak: this.currentStreak,
   797→   797→        lastCompletedDate: this.lastCompletedDate,
   798→   798→        durations: this.durations,
   799→   799→        sessionsUntilLong: this.sessionsUntilLong,
   800→   800→        autoStartBreaks: this.autoStartBreaks,
   801→   801→        autoStartWork: this.autoStartWork,
   802→   802→        soundEnabled: this.soundEnabled,
   803→   803→        taskText: this.elements.activeTaskText?.textContent || ''
   804→   804→      };
   805→   805→      
   806→   806→      try {
   807→   807→        localStorage.setItem('pomodoro-state', JSON.stringify(state));
   808→   808→      } catch (e) {
   809→   809→        console.log('Could not save state:', e);
   810→   810→      }
   811→   811→    }
   812→   812→    
   813→   813→    loadState() {
   814→   814→      try {
   815→   815→        const saved = localStorage.getItem('pomodoro-state');
   816→   816→        if (saved) {
   817→   817→          const state = JSON.parse(saved);
   818→   818→          
   819→   819→          // Only restore if saved today
   820→   820→          if (state.lastCompletedDate) {
   821→   821→            const savedDate = new Date(state.lastCompletedDate);
   822→   822→            const today = new Date();
   823→   823→            if (savedDate.toDateString() !== today.toDateString()) {
   824→   824→              // New day - reset daily stats but keep settings
   825→   825→              this.durations = state.durations || this.durations;
   826→   826→              this.sessionsUntilLong = state.sessionsUntilLong || this.sessionsUntilLong;
   827→   827→              this.autoStartBreaks = state.autoStartBreaks ?? this.autoStartBreaks;
   828→   828→              this.autoStartWork = state.autoStartWork ?? this.autoStartWork;
   829→   829→              this.soundEnabled = state.soundEnabled ?? this.soundEnabled;
   830→   830→              return;
   831→   831→            }
   832→   832→          }
   833→   833→          
   834→   834→          Object.assign(this, state);
   835→   835→          
   836→   836→          // Restore UI after a short delay for DOM
   837→   837→          setTimeout(() => {
   838→   838→            this.updateDisplay();
   839→   839→            this.updateSessionDots();
   840→   840→            this.updateStats();
   841→   841→            this.updatePlayButton();
   842→   842→            
   843→   843→            // Restore task
   844→   844→            if (state.taskText && this.elements.activeTaskText) {
   845→   845→              this.elements.activeTaskText.textContent = state.taskText;
   846→   846→              if (this.elements.activeTask) {
   847→   847→                this.elements.activeTask.style.display = 'flex';
   848→   848→              }
   849→   849→            }
   850→   850→            
   851→   851→            // Restore settings UI
   852→   852→            if (this.elements.workDuration) this.elements.workDuration.textContent = this.durations.work / 60;
   853→   853→            if (this.elements.breakDuration) this.elements.breakDuration.textContent = this.durations.break / 60;
   854→   854→            if (this.elements.longBreakDuration) this.elements.longBreakDuration.textContent = this.durations.longBreak / 60;
   855→   855→            if (this.elements.sessionsCount) this.elements.sessionsCount.textContent = this.sessionsUntilLong;
   856→   856→            if (this.elements.autoStartBreaksToggle) this.elements.autoStartBreaksToggle.checked = this.autoStartBreaks;
   857→   857→            if (this.elements.autoStartWorkToggle) this.elements.autoStartWorkToggle.checked = this.autoStartWork;
   858→   858→            if (this.elements.soundEnabledToggle) this.elements.soundEnabledToggle.checked = this.soundEnabled;
   859→   859→            
   860→   860→            // Update mode tabs
   861→   861→            this.elements.modeTabs.forEach(tab => {
   862→   862→              tab.classList.toggle('active', tab.dataset.mode === this.mode);
   863→   863→            });
   864→   864→            
   865→   865→            // Update progress color
   866→   866→            if (this.elements.timerProgress) {
   867→   867→              this.elements.timerProgress.style.stroke = this.getModeColor(this.mode);
   868→   868→            }
   869→   869→            
   870→   870→            // Resume if was running
   871→   871→            if (this.isRunning) {
   872→   872→              this.start();
   873→   873→            }
   874→   874→          }, 100);
   875→   875→        }
   876→   876→      } catch (e) {
   877→   877→        console.log('Could not load state:', e);
   878→   878→      }
   879→   879→      
   880→   880→      // Request notification permission on first interaction
   881→   881→      document.addEventListener('click', () => this.requestNotificationPermission(), { once: true });
   882→   882→    }
   883→   883→    
   884→   884→    // Helper to get i18n strings from data attributes
   885→   885→    getI18n(key) {
   886→   886→      const el = document.querySelector(`[data-i18n="${key}"]`);
   887→   887→      return el?.textContent || null;
   888→   888→    }
