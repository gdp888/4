   257→<script define:vars={{ lang }}>
   258→  // DOM Elements
   259→  const form = document.getElementById('scheduler-form');
   260→  const phoneInput = document.getElementById('phone');
   261→  const messageInput = document.getElementById('message');
   262→  const dateInput = document.getElementById('date');
   263→  const timeInput = document.getElementById('time');
   264→  const charCount = document.getElementById('char-count');
   265→  const submitBtn = document.getElementById('submit-btn');
   266→  const successMessage = document.getElementById('success-message');
   267→  const successDetails = document.getElementById('success-details');
   268→  const taskLink = document.getElementById('task-link');
   269→  const copyLinkBtn = document.getElementById('copy-link-btn');
   270→  const scheduleAnotherBtn = document.getElementById('schedule-another');
   271→  const tasksList = document.getElementById('tasks-list');
   272→  const timezoneDisplay = document.getElementById('timezone-display');
   273→
   274→  // Set minimum date to today
   275→  const today = new Date().toISOString().split('T')[0];
   276→  dateInput.setAttribute('min', today);
   277→
   278→  // Display user's timezone
   279→  const userTimezone = Intl.DateTimeFormat().resolvedOptions().timeZone;
   280→  timezoneDisplay.textContent = `Timezone: ${userTimezone}`;
   281→
   282→  // Character counter
   283→  messageInput.addEventListener('input', () => {
   284→    charCount.textContent = messageInput.value.length;
   285→  });
   286→
   287→  // Format phone number as user types
   288→  phoneInput.addEventListener('input', (e) => {
   289→    let value = e.target.value.replace(/\D/g, '');
   290→    if (value.length > 15) value = value.slice(0, 15);
   291→    e.target.value = value;
   292→  });
   293→
   294→  // Generate unique task ID client-side
   295→  function generateTaskId() {
   296→    const timestamp = Date.now().toString(36);
   297→    const randomPart = Math.random().toString(36).substring(2, 10);
   298→    return `task_${timestamp}_${randomPart}`;
   299→  }
   300→
   301→  // Form submission - with localStorage fallback
   302→  form.addEventListener('submit', async (e) => {
   303→    e.preventDefault();
   304→    
   305→    const phone = phoneInput.value.replace(/\D/g, '');
   306→    const message = messageInput.value.trim();
   307→    const date = dateInput.value;
   308→    const time = timeInput.value;
   309→
   310→    // Validate
   311→    if (phone.length < 8 || phone.length > 15) {
   312→      alert(lang === 'ru' ? 'Введите корректный номер телефона' : 'Please enter a valid phone number');
   313→      return;
   314→    }
   315→
   316→    const scheduledDateTime = new Date(`${date}T${time}`);
   317→    if (scheduledDateTime <= new Date()) {
   318→      alert(lang === 'ru' ? 'Выберите будущую дату и время' : 'Please select a future date and time');
   319→      return;
   320→    }
   321→
   322→    // Show loading state
   323→    submitBtn.disabled = true;
   324→    submitBtn.innerHTML = '<span class="spinner"></span> Scheduling...';
   325→
   326→    // Create task object (used for both API and localStorage)
   327→    const taskId = generateTaskId();
   328→    const task = {
   329→      id: taskId,
   330→      phone: phone,
   331→      message: message,
   332→      scheduledFor: scheduledDateTime.toISOString(),
   333→      timezone: userTimezone,
   334→      lang: lang,
   335→      status: 'scheduled',
   336→      createdAt: new Date().toISOString(),
   337→      whatsappLink: `https://wa.me/${phone}${message ? '?text=' + encodeURIComponent(message) : ''}`
   338→    };
   339→
   340→    try {
   341→      // Try API first (for future server-side implementation)
   342→      const response = await fetch('/api/scheduler', {
   343→        method: 'POST',
   344→        headers: { 'Content-Type': 'application/json' },
   345→        body: JSON.stringify({
   346→          phone,
   347→          message,
   348→          scheduledFor: scheduledDateTime.toISOString(),
   349→          timezone: userTimezone,
   350→          lang
   351→        })
   352→      });
   353→
   354→      if (response.ok) {
   355→        const data = await response.json();
   356→        if (data.success) {
   357→          // Use server-returned task data
   358→          Object.assign(task, data.task);
   359→        }
   360→      }
   361→    } catch (error) {
   362→      // API not available - continue with localStorage only
   363→      console.log('API not available, using localStorage mode');
   364→    }
   365→
   366→    // Save to localStorage (always, as backup/source of truth for now)
   367→    saveTaskLocally(task);
   368→
   369→    // Show success
   370→    form.classList.add('hidden');
   371→    successMessage.classList.remove('hidden');
   372→    
   373→    const formattedDate = new Date(task.scheduledFor).toLocaleString(lang);
   374→    successDetails.textContent = `+${task.phone.replace(/(\d{1,3})(\d{3})(\d{3})(\d{2,})/, '$1 $2 $3 $4')} at ${formattedDate}`;
   375→    
   376→    taskLink.href = `/scheduler/task/${task.id}`;
   377→    taskLink.textContent = `${window.location.origin}/scheduler/task/${task.id}`;
   378→    
   379→    // Refresh tasks list
   380→    loadTasks();
   381→
   382→    submitBtn.disabled = false;
   383→    submitBtn.innerHTML = `<svg class="btn-icon" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2">
   384→      <rect x="3" y="4" width="18" height="18" rx="2"/>
   385→      <line x1="16" y1="2" x2="16" y="6"/>
   386→      <line x1="8" y1="2" x2="8" y="6"/>
   387→      <line x1="3" y1="10" x2="21" y="10"/>
   388→    </svg> ${lang === 'ru' ? 'Запланировать сообщение' : 'Schedule Message'}`;
   389→  });
   390→
   391→  // Copy link button
   392→  copyLinkBtn.addEventListener('click', async () => {
   393→    try {
   394→      await navigator.clipboard.writeText(taskLink.href);
   395→      copyLinkBtn.textContent = 'Copied!';
   396→      setTimeout(() => {
   397→        copyLinkBtn.textContent = copyLinkBtn.dataset.copy;
   398→      }, 2000);
   399→    } catch (err) {
   400→      console.error('Failed to copy:', err);
   401→    }
   402→  });
   403→
   404→  // Schedule another button
   405→  scheduleAnotherBtn.addEventListener('click', () => {
   406→    form.reset();
   407→    charCount.textContent = '0';
   408→    form.classList.remove('hidden');
   409→    successMessage.classList.add('hidden');
   410→  });
   411→
   412→  // Local storage functions for tasks
   413→  function saveTaskLocally(task) {
   414→    const tasks = JSON.parse(localStorage.getItem('opnchat-scheduled-tasks') || '[]');
   415→    tasks.unshift(task);
   416→    // Keep only last 50 tasks
   417→    if (tasks.length > 50) tasks.pop();
   418→    localStorage.setItem('opnchat-scheduled-tasks', JSON.stringify(tasks));
   419→  }
   420→
   421→  function loadTasks() {
   422→    const tasks = JSON.parse(localStorage.getItem('opnchat-scheduled-tasks') || '[]');
   423→    
   424→    if (tasks.length === 0) {
   425→      tasksList.innerHTML = `
   426→        <div class="empty-state">
   427→          <svg class="empty-icon" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="1.5">
   428→            <rect x="3" y="4" width="18" height="18" rx="2"/>
   429→            <line x1="16" y1="2" x2="16" y2="6"/>
   430→            <line x1="8" y1="2" x2="8" y2="6"/>
   431→            <line x1="3" y1="10" x2="21" y2="10"/>
   432→            <line x1="9" y1="16" x2="15" y2="16"/>
   433→          </svg>
   434→          <p>No scheduled messages yet</p>
   435→        </div>
   436→      `;
   437→      return;
   438→    }
   439→
   440→    tasksList.innerHTML = tasks.map(task => {
   441→      const scheduledDate = new Date(task.scheduledFor);
   442→      const now = new Date();
   443→      const isPast = scheduledDate <= now;
   444→      const statusClass = isPast ? 'status-past' : 'status-pending';
   445→      const statusText = isPast ? 'Ready to send' : 'Scheduled';
   446→      
   447→      return `
   448→        <div class="task-item ${statusClass}">
   449→          <div class="task-info">
   450→            <span class="task-phone">${task.phone.replace(/(\d{1,3})(\d{3})(\d{3})(\d{2,})/, '+$1 $2 $3 $4')}</span>
   451→            <span class="task-time">${scheduledDate.toLocaleString(lang)}</span>
   452→          </div>
   453→          <div class="task-status">${statusText}</div>
   454→          <a href="/scheduler/task/${task.id}" class="task-view-btn" target="_blank">View</a>
   455→        </div>
   456→      `;
   457→    }).join('');
   458→  }
   459→
   460→  // Load tasks on page init
   461→  loadTasks();
   462→</script>
