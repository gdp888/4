  // Элементы DOM
  const form = document.getElementById('whatsapp-form');
  const countrySelect = document.getElementById('country-code');
  const phoneInput = document.getElementById('phone');
  const messageInput = document.getElementById('message');
  const errorMsg = document.getElementById('error-msg');
  
  const historySection = document.getElementById('history-section');
  const historyList = document.getElementById('history-list');

  // Инициализация истории из localStorage при загрузке страницы
  let historyData = JSON.parse(localStorage.getItem('opnchat_history') || '[]');
  renderHistory();

  // Обработчик отправки формы
  form.addEventListener('submit', (e) => {
    e.preventDefault();
    errorMsg.textContent = ''; // Сброс ошибок

    const countryCode = countrySelect.value.replace('+', ''); // убираем плюс, если он есть
    let localNumber = phoneInput.value.replace(/\D/g, ''); // оставляем только цифры

    // Простая валидация длины локального номера
    if (localNumber.length < 4) {
      errorMsg.textContent = 'Please enter a valid phone number.';
      return;
    }

    const fullPhoneNumber = countryCode + localNumber;
    const messageText = encodeURIComponent(messageInput.value.trim());

    // Формируем официальную безопасную ссылку WhatsApp Click-to-Chat
    let whatsappUrl = `https://wa.me{fullPhoneNumber}`;
    if (messageText) {
      whatsappUrl += `?text=${messageText}`;
    }

    // Сохраняем в локальную историю браузера
    saveToHistory(countrySelect.value, phoneInput.value.trim());

    // Перенаправляем пользователя в WhatsApp в новой вкладке
    window.open(whatsappUrl, '_blank', 'noopener,noreferrer');
  });

  // Функция сохранения номера в историю
  function saveToHistory(code, number) {
    const cleanNumber = number.replace(/\s+/g, ' '); // нормализуем пробелы
    const newItem = { code, number: cleanNumber, timestamp: Date.now() };

    // Проверяем на дубликаты, чтобы не захламлять список
    historyData = historyData.filter(item => !(item.code === code && item.number === cleanNumber));
    
    // Добавляем новый элемент в начало массива
    historyData.unshift(newItem);

    // Ограничиваем историю, например, последними 5 номерами
    if (historyData.length > 5) {
      historyData.pop();
    }

    localStorage.setItem('opnchat_history', JSON.stringify(historyData));
    renderHistory();
  }

  // Функция отрисовки интерфейса истории
  function renderHistory() {
    if (!historySection || !historyList) return;

    if (historyData.length === 0) {
      historySection.style.display = 'none';
      return;
    }

    historyList.innerHTML = '';
    historySection.style.display = 'block';

    historyData.forEach(item => {
      const li = document.createElement('li');
      li.className = 'history-item';
      li.style.cssText = 'display: flex; justify-content: space-between; align-items: center; margin-bottom: 8px; font-size: 14px;';
      
      // Контент с номером
      const numSpan = document.createElement('span');
      numSpan.textContent = `${item.code} ${item.number}`;
      numSpan.style.cursor = 'pointer';
      numSpan.style.fontWeight = '500';
      
      // При клике на номер из истории — автоподставляем его в форму
      numSpan.addEventListener('click', () => {
        countrySelect.value = item.code;
        phoneInput.value = item.number;
        phoneInput.focus();
      });

      // Кнопка быстрого вызова чата для этого номера
      const actionBtn = document.createElement('button');
      actionBtn.type = 'button';
      actionBtn.className = 'history-btn';
      actionBtn.textContent = 'Chat';
      actionBtn.style.cssText = 'padding: 4px 12px; font-size: 12px; cursor: pointer; border-radius: 4px; background: #0052cc; color: white; border: none;';
      
      actionBtn.addEventListener('click', () => {
        const cleanCode = item.code.replace('+', '');
        const cleanNum = item.number.replace(/\D/g, '');
        window.open(`https://wa.me{cleanCode}${cleanNum}`, '_blank', 'noopener,noreferrer');
      });

      li.appendChild(numSpan);
      li.appendChild(actionBtn);
      historyList.appendChild(li);
    });
  }
